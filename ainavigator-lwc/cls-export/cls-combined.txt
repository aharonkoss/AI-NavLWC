/**
 * AI_Navigator_Event_Publisher
 *
 * Shared utility for publishing AI_Navigator_Outbound__e Platform Events.
 * All outbound service classes call this — zero duplication of
 * event construction, config loading, or error handling.
 *
 * Reads config from AI_Navigator_Config__mdt (DeveloperName = 'Dev').
 * Kill switch: if Is_Active__c = false, publish is skipped entirely.
 */
public without sharing class AI_Navigator_Event_Publisher {

    // ── Publish a single outbound event ──────────────────────────────────────
    public static void publish(
        String eventType,
        String msgType,
        String schemaName,
        String payloadJson
    ) {
        publish(eventType, msgType, schemaName, payloadJson, 'NEW');
    }

    // ── Publish with explicit status (NEW or RETRY) ──────────────────────────
    public static void publish(
        String eventType,
        String msgType,
        String schemaName,
        String payloadJson,
        String statusCode
    ) {
        // Load config — fail fast if missing
        AI_Navigator_Config__mdt config = loadConfig();
        if (config == null) {
            System.debug(LoggingLevel.ERROR,
                'AI_Navigator_Event_Publisher: No active config record found. ' +
                'Create an AI_Navigator_Config__mdt record with DeveloperName = Dev.');
            return;
        }

        // Kill switch check
        if (!config.Is_Active__c) {
            System.debug(LoggingLevel.WARN,
                'AI_Navigator_Event_Publisher: Publishing disabled via kill switch. ' +
                'Set Is_Active__c = true on AI_Navigator_Config__mdt to re-enable.');
            return;
        }

        // Validate payload size — Platform Events cap at 32,000 chars
        if (String.isNotBlank(payloadJson) && payloadJson.length() > 32000) {
            System.debug(LoggingLevel.ERROR,
                'AI_Navigator_Event_Publisher: Payload exceeds 32,000 char limit ' +
                'for Event_Type: ' + eventType);
            return;
        }

        // Build the event
        AI_Navigator_Outbound__e evt = new AI_Navigator_Outbound__e(
            Message_Id__c      = generateUUID(),
            Event_Type__c      = eventType,
            Msg_Type__c        = msgType,
            Status_Code__c     = statusCode,
            Client_Id__c       = config.Client_Id__c,
            User_Id__c         = UserInfo.getUserId(),
            Correlation_Id__c  = generateUUID(),
            Content_Type__c    = 'json',
            Schema_Name__c     = schemaName + '.' + config.Event_Schema_Version__c,
            Payload__c         = payloadJson,
            Created_At__c      = DateTime.now()
        );

        // Publish
        Database.SaveResult result = EventBus.publish(evt);

        if (!result.isSuccess()) {
            for (Database.Error err : result.getErrors()) {
                System.debug(LoggingLevel.ERROR,
                    'AI_Navigator_Event_Publisher: Publish failed for ' + eventType +
                    ' | ' + err.getStatusCode() + ': ' + err.getMessage());
            }
        } else {
            System.debug(LoggingLevel.INFO,
                'AI_Navigator_Event_Publisher: Published ' + eventType +
                ' | Correlation: ' + evt.Correlation_Id__c);
        }
    }

    // ── Private: load config ─────────────────────────────────────────────────
    private static AI_Navigator_Config__mdt loadConfig() {
        List<AI_Navigator_Config__mdt> configs = [
            SELECT Client_Id__c, Azure_APIM_Base_URL__c, Named_Credential_Name__c,
                   Max_Retry_Attempts__c, Callout_Timeout_Ms__c,
                   Event_Schema_Version__c, Is_Active__c
            FROM AI_Navigator_Config__mdt
            WHERE DeveloperName = 'Dev'
            LIMIT 1
        ];
        return configs.isEmpty() ? null : configs[0];
    }

    // ── Private: generate UUID v4 ────────────────────────────────────────────
    // Salesforce has no native UUID generator so we build one from Crypto
    public static String generateUUID() {
        Blob b = Crypto.generateAesKey(128);
        String hex = EncodingUtil.convertToHex(b);
        return hex.substring(0,8)  + '-' +
               hex.substring(8,12) + '-' +
               '4' + hex.substring(13,16) + '-' +
               hex.substring(16,20) + '-' +
               hex.substring(20,32);
    }
        /**
     * Generate a correlation ID in the format Azure APIM requires:
     * SF-<unix-epoch-ms>-<8-char-hex>
     * Example: SF-1777404715585-50cd8363
     */
        public static String generateCorrelationId() {
            String unixMs  = String.valueOf(DateTime.now().getTime());
            String hexPart = EncodingUtil.convertToHex(Crypto.generateAesKey(128)).substring(0, 8);
            return 'SF-' + unixMs + '-' + hexPart;
        }
}

/**
 * AI_Navigator_Inbound_Handler
 *
 * Routes inbound Platform Events from Azure to the correct
 * processor based on Event_Type__c. All DML is bulkified —
 * the trigger may fire with multiple events at once.
 *
 * Supported Event Types:
 *   RESEARCH_DONE      → CompanyStatusUpdater
 *   UPLOAD_PROCESSED   → SubmissionProgressUpdater
 *   BATCH_PROGRESS     → SubmissionProgressUpdater
 *   SIGNAL_ALERT       → SignalAlertProcessor
 *   SCORING_DONE       → CompanyStatusUpdater
 *   REPORT_READY       → CompanyStatusUpdater
 *   PROCESSING_ERROR   → CompanyStatusUpdater
 */
public without sharing class AI_Navigator_Inbound_Handler {

    // ── Entry Point ──────────────────────────────────────────────────────────
    public static void handle(List<AI_Navigator_Inbound__e> events) {

        // Partition events by type for bulkified processing
        List<AI_Navigator_Inbound__e> researchEvents    = new List<AI_Navigator_Inbound__e>();
        List<AI_Navigator_Inbound__e> uploadEvents      = new List<AI_Navigator_Inbound__e>();
        List<AI_Navigator_Inbound__e> progressEvents    = new List<AI_Navigator_Inbound__e>();
        List<AI_Navigator_Inbound__e> signalEvents      = new List<AI_Navigator_Inbound__e>();
        List<AI_Navigator_Inbound__e> scoringEvents     = new List<AI_Navigator_Inbound__e>();
        List<AI_Navigator_Inbound__e> reportEvents      = new List<AI_Navigator_Inbound__e>();
        List<AI_Navigator_Inbound__e> errorEvents       = new List<AI_Navigator_Inbound__e>();

        for (AI_Navigator_Inbound__e evt : events) {

            // Log every inbound event to the request log
            logEvent(evt);

            if (String.isBlank(evt.Event_Type__c)) {
                System.debug(LoggingLevel.WARN,
                    'AI_Navigator_Inbound_Handler: Received event with no Event_Type__c. ' +
                    'Message_Id: ' + evt.Message_Id__c);
                continue;
            }

            switch on evt.Event_Type__c {
                when 'RESEARCH_DONE'    { researchEvents.add(evt); }
                when 'UPLOAD_PROCESSED' { uploadEvents.add(evt); }
                when 'BATCH_PROGRESS'   { progressEvents.add(evt); }
                when 'SIGNAL_ALERT'     { signalEvents.add(evt); }
                when 'SCORING_DONE'     { scoringEvents.add(evt); }
                when 'REPORT_READY'     { reportEvents.add(evt); }
                when 'PROCESSING_ERROR' { errorEvents.add(evt); }
                when else {
                    System.debug(LoggingLevel.WARN,
                        'AI_Navigator_Inbound_Handler: Unknown Event_Type__c: ' +
                        evt.Event_Type__c + ' | Message_Id: ' + evt.Message_Id__c);
                }
            }
        }

        // ── Dispatch to processors ───────────────────────────────────────────
        if (!researchEvents.isEmpty()) {
            CompanyStatusUpdater.processResearchDone(researchEvents);
        }
        if (!uploadEvents.isEmpty()) {
            SubmissionProgressUpdater.processUploadComplete(uploadEvents);
        }
        if (!progressEvents.isEmpty()) {
            SubmissionProgressUpdater.processBatchProgress(progressEvents);
        }
        if (!signalEvents.isEmpty()) {
            SignalAlertProcessor.processSignalAlerts(signalEvents);
        }
        if (!scoringEvents.isEmpty()) {
            CompanyStatusUpdater.processScoringDone(scoringEvents);
        }
        if (!reportEvents.isEmpty()) {
            CompanyStatusUpdater.processReportReady(reportEvents);
        }
        if (!errorEvents.isEmpty()) {
            CompanyStatusUpdater.processError(errorEvents);
        }
    }

    // ── Private: log every event to Azure_Request_Log__c ────────────────────
    private static void logEvent(AI_Navigator_Inbound__e evt) {
        try {
            Azure_Request_Log__c log = new Azure_Request_Log__c(
                HTTP_Method__c      = 'PLATFORM_EVENT',
                Endpoint_Name__c    = 'AI_Navigator_Inbound__e',
                Request_Body__c     = evt.Payload__c,
                Status__c           = evt.Status_Code__c,
                Correlation_ID__c   = evt.Correlation_Id__c,
                Error_Message__c    = evt.Event_Type__c + ' | ' + evt.Schema_Name__c
            );
            insert log;
        } catch (DmlException de) {
            System.debug(LoggingLevel.ERROR,
                'AI_Navigator_Inbound_Handler: Failed to write log. ' + de.getMessage());
        }
    }
}

/**
 * AI_Navigator_Outbound_Handler
 *
 * Routes outbound platform events to the correct Azure HTTP callout
 * based on EventType__c. Runs as a Queueable to allow HTTP callouts
 * from a Platform Event trigger context.
 */
public with sharing class AI_Navigator_Outbound_Handler {

    public static void handle(List<AI_Navigator_Outbound__e> events) {
        for (AI_Navigator_Outbound__e evt : events) {
            if (evt.Event_Type__c == 'UPLOAD') {
                System.enqueueJob(new UploadCalloutJob(evt.Payload__c, evt.Correlation_ID__c));
            }
            // Future event types (RESEARCH_REQUEST, etc.) get else-if branches here
        }
    }

    // ── Queueable: makes the actual HTTP callout to Azure ──────────────────
    public class UploadCalloutJob implements Queueable, Database.AllowsCallouts {

        private String payload;
        private String correlationId;

        public UploadCalloutJob(String payload, String correlationId) {
            this.payload     = payload;
            this.correlationId = correlationId;
        }

        public void execute(QueueableContext ctx) {
            HttpRequest req = new HttpRequest();
            req.setEndpoint('callout:Azure_AI_Navigator/api/v1/upload');
            req.setMethod('POST');
            req.setHeader('Content-Type', 'application/json');
            req.setHeader('X-Correlation-Id', correlationId);
            req.setBody(payload);
            req.setTimeout(120000); // 2 minutes

            HttpResponse res = new Http().send(req);

            writeLog(res, correlationId);
        }

        private void writeLog(HttpResponse res, String corrId) {
            String status  = (res.getStatusCode() >= 200 && res.getStatusCode() < 300) ? 'Success' : 'Failed';
            try {
                insert new Azure_Request_Log__c(
                    HTTP_Method__c      = 'POST',
                    Endpoint_Name__c    = 'AI_Navigator_Outbound_Handler.UploadCalloutJob',
                    Response_Body__c    = res.getBody(),
                    Status__c           = status,
                    HTTP_Status_Code__c = res.getStatusCode(),
                    Correlation_ID__c   = corrId,
                    Error_Message__c    = status == 'Failed' ? 'Azure returned: ' + res.getStatus() : null
                );
            } catch (Exception e) {
                System.debug(LoggingLevel.ERROR, 'writeLog failed: ' + e.getMessage());
            }
        }
    }
}

/**
 * @description Shared service for publishing AI_Navigator_Outbound__e Platform Events.
 *              All outbound event publishing flows through this class so schema versioning
 *              and correlation ID generation are centralized.
 *
 * @group AI Navigator Integration
 * @last-modified 2026-04-27
 */
public with sharing class AiNavigatorEventService {

    /**
     * @description Publishes a single AI_Navigator_Outbound__e event.
     *
     * @param eventType    The event type string (e.g. 'RESEARCH_REQUEST', 'UPLOAD')
     * @param schemaName   The payload schema version (e.g. 'ResearchRequest.v1')
     * @param payload      Map of fields to serialize as the JSON payload
     * @param correlationId Caller-supplied correlation ID already stamped on the SObject
     */
    public static void publish(
        String eventType,
        String schemaName,
        Map<String, Object> payload,
        String correlationId
    ) {
        // Always embed correlationId and schemaVersion inside the payload itself
        // so Azure has everything it needs without reading envelope fields separately.
        payload.put('correlationId',  correlationId);
        payload.put('schemaVersion',  schemaName);

        AI_Navigator_Outbound__e evt = new AI_Navigator_Outbound__e(
            Event_Type__c    = eventType,
            Schema_Name__c   = schemaName,
            Payload__c      = JSON.serialize(payload),
            Correlation_Id__c = correlationId
        );

        Database.SaveResult sr = EventBus.publish(evt);
        if (!sr.isSuccess()) {
            String errMsg = sr.getErrors()[0].getMessage();
            System.debug(LoggingLevel.ERROR,
                'AiNavigatorEventService: Failed to publish ' + eventType
                + ' | correlationId=' + correlationId + ' | error=' + errMsg);
            throw new AuraHandledException(
                'Failed to send request to processing pipeline. Please try again. ' + errMsg
            );
        }
        System.debug(LoggingLevel.INFO,
            'AiNavigatorEventService: Published ' + eventType
            + ' | schema=' + schemaName + ' | correlationId=' + correlationId);
    }

    /**
     * @description Generates a Salesforce-owned correlation ID in the format
     *              Azure expects for end-to-end trace correlation.
     *
     *   Format  : SF-{unix-ms}-{8-char-lowercase-hex}
     *   Example : SF-1714060800000-a3f9d2c1
     *
     *   The 8-char hex tail provides ~4 billion values per millisecond bucket —
     *   sufficient entropy for trace correlation per Azure spec v2.
     *
     *   CHANGE FROM PREVIOUS VERSION:
     *     Old: 4-char uppercase hex  → SF-1745271600000-A3F2
     *     New: 8-char lowercase hex  → SF-1714060800000-a3f9d2c1
     *     Reason: Azure developer confirmed format requirement 2026-04-27.
     */
    public static String generateCorrelationId() {
        String ts   = String.valueOf(DateTime.now().getTime());
        String rand = EncodingUtil.convertToHex(
                          Crypto.generateAesKey(128)
                      ).substring(0, 8).toLowerCase();
        return 'SF-' + ts + '-' + rand;
    }
}

/**
 * @description Unit tests for AiNavigatorEventService.
 *              Covers correlation ID format contract (v2) and publish happy-path.
 *
 * @group AI Navigator - Tests
 */
@IsTest
private class AiNavigatorEventServiceTest {

    // -------------------------------------------------------------------------
    // generateCorrelationId — format contract
    // -------------------------------------------------------------------------

    @IsTest
    static void testCorrelationIdFormat() {
        Test.startTest();
        String corrId = AiNavigatorEventService.generateCorrelationId();
        Test.stopTest();

        // Must start with 'SF-'
        System.assert(corrId.startsWith('SF-'),
            'Correlation ID must start with SF- but was: ' + corrId);

        // Split into three parts: prefix, timestamp-ms, hex
        List<String> parts = corrId.split('-', 3);   // limit 3 so hex isn't split
        System.assertEquals(3, parts.size(),
            'Expected 3 dash-separated parts but got: ' + corrId);

        String prefix    = parts[0];   // 'SF'
        String tsSegment = parts[1];   // unix-ms digits
        String hexSegment = parts[2];  // 8-char lowercase hex

        // Prefix
        System.assertEquals('SF', prefix, 'First segment must be SF');

        // Timestamp segment must be all digits and 13 chars (unix-ms)
        System.assert(tsSegment.isNumeric(),
            'Timestamp segment must be numeric but was: ' + tsSegment);
        System.assertEquals(13, tsSegment.length(),
            'Timestamp segment must be 13 digits (unix-ms) but was length: ' + tsSegment.length());

        // Hex segment must be exactly 8 chars, all lowercase hex
        System.assertEquals(8, hexSegment.length(),
            'Hex segment must be 8 chars but was: ' + hexSegment);
        System.assert(hexSegment.equals(hexSegment.toLowerCase()),
            'Hex segment must be lowercase but was: ' + hexSegment);
        System.assert(hexSegment.containsOnly('0123456789abcdef'),
            'Hex segment must contain only hex chars but was: ' + hexSegment);

        // Total length must fit Text(36) field
        System.assert(corrId.length() <= 36,
            'Correlation ID length ' + corrId.length() + ' exceeds Text(36): ' + corrId);
    }

    @IsTest
    static void testCorrelationIdIsUniqueAcrossCalls() {
        Test.startTest();
        String id1 = AiNavigatorEventService.generateCorrelationId();
        String id2 = AiNavigatorEventService.generateCorrelationId();
        Test.stopTest();

        System.assertNotEquals(id1, id2,
            'Two consecutive correlation IDs must not be identical');
    }

    // -------------------------------------------------------------------------
    // publish — happy path (event bus mock)
    // -------------------------------------------------------------------------

    @IsTest
    static void testPublishSucceeds() {
        Test.startTest();
        String corrId = AiNavigatorEventService.generateCorrelationId();
        Map<String, Object> payload = new Map<String, Object>{
            'companyName' => 'Acme Corp',
            'website'     => 'https://www.acme.com'
        };

        // EventBus.publish is allowed in @IsTest context; it doesn't actually fire
        // but the SaveResult is successful so no exception should be thrown.
        Boolean threw = false;
        try {
            AiNavigatorEventService.publish(
                'RESEARCH_REQUEST',
                'ResearchRequest.v1',
                payload,
                corrId
            );
        } catch (Exception e) {
            threw = true;
        }
        Test.stopTest();

        System.assert(!threw, 'publish() should not throw on a well-formed event');

        // Verify correlationId and schemaVersion were injected into the payload map
        System.assertEquals(corrId, payload.get('correlationId'),
            'correlationId must be embedded in the payload map');
        System.assertEquals('ResearchRequest.v1', payload.get('schemaVersion'),
            'schemaVersion must be embedded in the payload map');
    }
}

/**
 * AzureCompanyCallbackService
 *
 * Direct HTTP callback endpoint. Azure calls this to update a
 * specific Company__c record with research results, scores, or
 * status changes without going through the Platform Event bus.
 * Used for lightweight updates where full event routing is overkill.
 *
 * Endpoint : /services/apexrest/ainavigator/v1/companies/callback
 * Method   : POST
 * Auth     : OAuth 2.0 Bearer Token (Client Credentials Flow)
 *
 * Expected body:
 * {
 *   "companyId"    : "a0Bxx000001SvgsEAC",   // Salesforce record ID
 *   "status"       : "completed",             // pending/processing/completed/failed
 *   "prospectScore": 78.5,                    // optional
 *   "pdfReportUrl" : "https://...",           // optional
 *   "correlationId": "uuid"                   // optional
 * }
 */
@RestResource(urlMapping='/ainavigator/v1/companies/callback')
global without sharing class AzureCompanyCallbackService {

    private static final String CORR_HEADER = 'X-Correlation-Id';

    @HttpPost
    global static void handlePost() {

        RestRequest  req = RestContext.request;
        RestResponse res = RestContext.response;

        String correlationId = '';
        String requestBody   = '';

        try {
            // ── 1. Headers ───────────────────────────────────────────────────
            correlationId = req.headers.get(CORR_HEADER);
            if (String.isBlank(correlationId)) {
                correlationId = 'NO-CORR-ID-' + DateTime.now().getTime();
            }

            // ── 2. Parse body ────────────────────────────────────────────────
            requestBody = req.requestBody != null ? req.requestBody.toString() : '';
            if (String.isBlank(requestBody)) {
                sendResponse(res, 400, 'error', 'Request body is required.', correlationId);
                return;
            }

            Map<String, Object> body;
            try {
                body = (Map<String, Object>) JSON.deserializeUntyped(requestBody);
            } catch (JSONException je) {
                sendResponse(res, 400, 'error', 'Invalid JSON.', correlationId);
                return;
            }

            // ── 3. Validate companyId ────────────────────────────────────────
            String companyId = (String) body.get('companyId');
            if (String.isBlank(companyId)) {
                sendResponse(res, 400, 'error', 'companyId is required.', correlationId);
                return;
            }

            // ── 4. Fetch company ─────────────────────────────────────────────
            List<Company__c> companies = [
                SELECT Id, Status__c, Prospect_Score__c, Scored_At__c,
                       PDF_Report_URL__c, Research_Completed_At__c,
                       Correlation_Id__c
                FROM Company__c
                WHERE Id = :companyId
                LIMIT 1
            ];

            if (companies.isEmpty()) {
                sendResponse(res, 404, 'error',
                    'Company not found: ' + companyId, correlationId);
                return;
            }

            Company__c company = companies[0];

            // ── 5. Apply updates from payload ────────────────────────────────
            String newStatus = (String) body.get('status');
            if (String.isNotBlank(newStatus)) {
                company.Status__c = newStatus;
                if (newStatus == 'completed' || newStatus == 'failed') {
                    company.Research_Completed_At__c = DateTime.now();
                }
            }

            Object scoreVal = body.get('prospectScore');
            if (scoreVal != null) {
                company.Prospect_Score__c = Decimal.valueOf(String.valueOf(scoreVal));
                company.Scored_At__c      = DateTime.now();
            }

            String pdfUrl = (String) body.get('pdfReportUrl');
            if (String.isNotBlank(pdfUrl)) {
                company.PDF_Report_URL__c = pdfUrl;
            }

            company.Correlation_Id__c = correlationId;

            // ── 6. Save ──────────────────────────────────────────────────────
            try {
                update company;
            } catch (DmlException de) {
                sendResponse(res, 500, 'error',
                    'Failed to update company record.', correlationId);
                return;
            }

            // ── 7. Log and respond ───────────────────────────────────────────
            logRequest(correlationId, companyId, requestBody, 200);

            sendResponse(res, 200, 'success',
                'Company record updated successfully.', correlationId);

        } catch (Exception e) {
            System.debug(LoggingLevel.ERROR,
                'AzureCompanyCallbackService: ' + e.getTypeName() +
                ': ' + e.getMessage() + ' | Line: ' + e.getLineNumber());
            sendResponse(res, 500, 'error',
                'Internal Server Error. Contact the Salesforce admin.', correlationId);
        }
    }

    // ── Private helpers ──────────────────────────────────────────────────────
    private static void sendResponse(
        RestResponse res,
        Integer      statusCode,
        String       status,
        String       message,
        String       correlationId
    ) {
        Map<String, Object> body = new Map<String, Object>{
            'status'        => status,
            'message'       => message,
            'correlationId' => correlationId,
            'timestamp'     => String.valueOf(DateTime.now())
        };
        res.statusCode   = statusCode;
        res.responseBody = Blob.valueOf(JSON.serializePretty(body));
        res.headers.put('Content-Type', 'application/json');
        res.headers.put('X-Correlation-Id', correlationId);
    }

    private static void logRequest(
        String correlationId,
        String companyId,
        String requestBody,
        Integer statusCode
    ) {
        try {
            insert new Azure_Request_Log__c(
                HTTP_Method__c      = 'POST',
                Endpoint_Name__c    = 'AzureCompanyCallbackService',
                Request_Body__c     = requestBody,
                Status__c           = statusCode == 200 ? 'Success' : 'Error',
                HTTP_Status_Code__c = statusCode,
                Correlation_ID__c   = correlationId,
                Error_Message__c    = 'CompanyId: ' + companyId
            );
        } catch (DmlException de) {
            System.debug(LoggingLevel.ERROR,
                'AzureCompanyCallbackService: Log failed. ' + de.getMessage());
        }
    }
}

/**
 * AzureEchoService
 *
 * Secure Apex REST endpoint for testing Azure → Salesforce connectivity.
 * Accepts any JSON POST body and returns it verbatim.
 * Logs every request (success and failure) to Azure_Request_Log__c.
 *
 * Endpoint : /services/apexrest/ainavigator/v1/echo
 * Method   : POST
 * Auth     : OAuth 2.0 Bearer Token (Client Credentials Flow)
 */
@RestResource(urlMapping='/ainavigator/v1/echo')
global with sharing class AzureEchoService {


    // ── Constants ────────────────────────────────────────────────────────────
    private static final String  ENDPOINT_NAME     = 'AzureEchoService';
    private static final Integer MAX_BODY_BYTES     = 32768;   // 32 KB hard limit
    private static final String  CORR_HEADER        = 'X-Correlation-ID';
    private static final String  CONTENT_TYPE_HDR   = 'Content-Type';


    // ── Entry Point ──────────────────────────────────────────────────────────
    @HttpPost
    global static void handlePost() {

        RestRequest  req = RestContext.request;
        RestResponse res = RestContext.response;

        String  correlationId = '';
        String  requestBody   = '';
        String  sourceIp      = '';
        Integer bodyBytes     = 0;

        try {
            // ── 1. Extract metadata ──────────────────────────────────────────
            correlationId = req.headers.get(CORR_HEADER);
            if (String.isBlank(correlationId)) {
                correlationId = 'NO-CORR-ID-' + String.valueOf(DateTime.now().getTime());
            }

            sourceIp    = req.remoteAddress;
            requestBody = req.requestBody != null ? req.requestBody.toString() : '';
            bodyBytes   = req.requestBody != null ? req.requestBody.size() : 0;

            // ── 2. Validate Content-Type ─────────────────────────────────────
            String contentType = req.headers.get(CONTENT_TYPE_HDR);
            if (contentType == null || !contentType.containsIgnoreCase('application/json')) {
                sendError(res, 415, 'Unsupported Media Type. Content-Type must be application/json.',
                          correlationId, requestBody, sourceIp, bodyBytes);
                return;
            }

            // ── 3. Enforce payload size limit (32 KB) ────────────────────────
            if (bodyBytes > MAX_BODY_BYTES) {
                sendError(res, 413, 'Payload Too Large. Maximum request body size is 32,768 bytes.',
                          correlationId, requestBody, sourceIp, bodyBytes);
                return;
            }

            // ── 4. Validate the body is parseable JSON ───────────────────────
            if (String.isNotBlank(requestBody)) {
                try {
                    JSON.deserializeUntyped(requestBody);
                } catch (JSONException je) {
                    sendError(res, 400, 'Bad Request. Request body is not valid JSON: ' + je.getMessage(),
                              correlationId, requestBody, sourceIp, bodyBytes);
                    return;
                }
            }

            // ── 5. Build echo response ───────────────────────────────────────
            Map<String, Object> echoPayload = new Map<String, Object>{
                'status'        => 'success',
                'correlationId' => correlationId,
                'echo'          => String.isNotBlank(requestBody)
                                   ? JSON.deserializeUntyped(requestBody)
                                   : new Map<String, Object>(),
                'receivedAt'    => String.valueOf(DateTime.now()),
                'endpoint'      => ENDPOINT_NAME
            };

            String responseBody = JSON.serializePretty(echoPayload);

            // ── 6. Set response ──────────────────────────────────────────────
            res.statusCode   = 200;
            res.responseBody = Blob.valueOf(responseBody);
            res.headers.put('Content-Type', 'application/json');
            res.headers.put(CORR_HEADER, correlationId);

            // ── 7. Log success (synchronous — no @future) ───────────────────
            logRequest(
                correlationId, 'POST', requestBody, responseBody,
                200, 'Success', null, sourceIp, bodyBytes
            );

        } catch (Exception e) {
            // ── 8. Catch-all — log and return 500 ───────────────────────────
            String errMsg = e.getTypeName() + ': ' + e.getMessage()
                            + ' | Line: ' + e.getLineNumber();
            sendError(res, 500, 'Internal Server Error. Contact the Salesforce admin.',
                      correlationId, requestBody, sourceIp, bodyBytes, errMsg);
        }
    }


    // ── Private: send error response + log ──────────────────────────────────
    private static void sendError(
        RestResponse res,
        Integer      statusCode,
        String       userMessage,
        String       correlationId,
        String       requestBody,
        String       sourceIp,
        Integer      bodyBytes
    ) {
        sendError(res, statusCode, userMessage, correlationId,
                  requestBody, sourceIp, bodyBytes, null);
    }

    private static void sendError(
        RestResponse res,

        String       userMessage,
        String       correlationId,
        String       requestBody,

            'status'        => 'error',

        res.responseBody = Blob.valueOf(responseBody);

        res.headers.put(CORR_HEADER, correlationId);

                           : 'Error';

        logRequest(
            correlationId, 'POST', requestBody, responseBody,
            statusCode, logStatus,

    // ── Private: write to Azure_Request_Log__c (synchronous DML) ────────────
    // ✅ Removed @future — DML now executes in the same transaction as the

/**
 * @description Shared HTTP utility for all outbound Azure APIM callouts.
 *              Builds pre-configured HttpRequest objects using headers pulled
 *              from AI_Navigator_Config__mdt (DeveloperName = 'Dev').
 *
 *              Required Azure headers:
 *                - Ocp-Apim-Subscription-Key  ← new ApimSubscriptionKey__c field
 *                - X-Client-Id                ← Client_Id__c from CMT
 *                - X-User-Id                  ← SF User Id of submitting user
 *                - X-Correlation-Id           ← UUID generated per request
 *                - Content-Type application/json
 *                - Authorization Bearer       ← injected by Named Credential
 */
public without sharing class AzureHttpService {

    private static final Integer DEFAULT_TIMEOUT_MS = 30000;

    /**
     * Build a ready-to-send HttpRequest with all required Azure headers.
     *
     * @param method           HTTP verb: POST, GET, PUT
     * @param path             Path appended to Named Credential base URL
     *                         e.g. /api/v1/user/companies/manual
     * @param correlationId    UUID generated by the caller
     * @param submittingUserId Salesforce User Id of the user who triggered the action
     * @return Fully configured HttpRequest — caller sets body if needed, then sends
     */
    public static HttpRequest buildRequest(
        String method,
        String path,
        String correlationId,
        String submittingUserId
    ) {
        AI_Navigator_Config__mdt config = loadConfig();

        HttpRequest req = new HttpRequest();

        // Named Credential handles Authorization: Bearer <JWT> automatically
        req.setEndpoint('callout:' + config.Named_Credential_Name__c + path);
        req.setMethod(method);
        req.setTimeout(
            config.Callout_Timeout_Ms__c != null
                ? Integer.valueOf(config.Callout_Timeout_Ms__c)
                : DEFAULT_TIMEOUT_MS
        );

        // Required headers — matched to Azure APIM contract
        req.setHeader('Content-Type',               'application/json');
        req.setHeader('Accept',                     'application/json');
        req.setHeader('Ocp-Apim-Subscription-Key',  config.ApimSubscriptionKey__c);
        req.setHeader('X-Client-Id',                config.Client_Id__c);
        req.setHeader('X-User-Id',
            String.isNotBlank(submittingUserId) ? submittingUserId : UserInfo.getUserId()
        );
        req.setHeader('X-Correlation-Id',           correlationId);

        return req;
    }

    /**
     * Parse a standard Azure error response envelope and return a readable message.
     */
    public static String parseErrorResponse(HttpResponse res) {
        try {
            Map<String, Object> body =
                (Map<String, Object>) JSON.deserializeUntyped(res.getBody());
            String msg    = (String) body.get('message');
            String code   = (String) body.get('code');
            String corrId = (String) body.get('correlationId');
            return (code   != null ? '[' + code   + '] ' : '')
                 + (msg    != null ? msg            : res.getBody())
                 + (corrId != null ? ' (correlationId: ' + corrId + ')' : '');
        } catch (Exception e) {
            return 'HTTP ' + res.getStatusCode() + ' — ' + res.getBody();
        }
    }

    // ── Private ──────────────────────────────────────────────────────────────

    private static AI_Navigator_Config__mdt loadConfig() {
        List<AI_Navigator_Config__mdt> configs = [
            SELECT Named_Credential_Name__c,
                   ApimSubscriptionKey__c,
                   Client_Id__c,
                   Callout_Timeout_Ms__c,
                   Is_Active__c
            FROM   AI_Navigator_Config__mdt
            WHERE  DeveloperName = 'Dev'
            LIMIT  1
        ];

        if (configs.isEmpty() || configs[0].Named_Credential_Name__c == null) {
            throw new AzureHttpServiceException(
                'AI_Navigator_Config__mdt "Dev" record is missing or incomplete.'
            );
        }
        if (!configs[0].Is_Active__c) {
            throw new AzureHttpServiceException(
                'Azure callouts are disabled via kill switch (Is_Active__c = false).'
            );
        }
        return configs[0];
    }

    public class AzureHttpServiceException extends Exception {}
}

/**
 * AzureInboundEventService
 *
 * Apex REST endpoint that accepts inbound event payloads from Azure
 * and publishes them as AI_Navigator_Inbound__e Platform Events.
 * This is the primary HTTP callback endpoint for Azure → Salesforce.
 *
 * Endpoint : /services/apexrest/ainavigator/v1/events/inbound
 * Method   : POST
 * Auth     : OAuth 2.0 Bearer Token (Client Credentials Flow)
 *
 * Azure posts this when async processing completes or alerts fire.
 * Salesforce publishes the Platform Event which triggers the
 * AI_Navigator_Inbound_Trigger → AI_Navigator_Inbound_Handler chain.
 */
@RestResource(urlMapping='/ainavigator/v1/events/inbound')
global without sharing class AzureInboundEventService {

    private static final String CORR_HEADER      = 'X-Correlation-Id';
    private static final String CLIENT_HDR       = 'X-Client-Id';
    private static final Integer MAX_BODY_BYTES  = 65536; // 64 KB

    @HttpPost
    global static void handlePost() {

        RestRequest  req = RestContext.request;
        RestResponse res = RestContext.response;

        String correlationId = '';
        String requestBody   = '';

        try {
            // ── 1. Extract headers ───────────────────────────────────────────
            // Headers are case-sensitive in RestContext — try both casings,
            // then fall back to the body field Correlation_Id__c.
            correlationId = req.headers.get('X-Correlation-Id');
            if (String.isBlank(correlationId)) {
                correlationId = req.headers.get('x-correlation-id');
            }
            if (String.isBlank(correlationId)) {
                correlationId = req.headers.get('X-CORRELATION-ID');
            }
            // Body field fallback — Azure also sends it as Correlation_Id__c
            // Parse body first so we can read it here
            String rawBody = req.requestBody != null ? req.requestBody.toString() : '';
            if (String.isBlank(correlationId) && String.isNotBlank(rawBody)) {
                try {
                    Map<String, Object> tempBody =
                        (Map<String, Object>) JSON.deserializeUntyped(rawBody);
                    correlationId = (String) tempBody.get('Correlation_Id__c');
                } catch (Exception e) { /* ignore — full parse happens in step 4 */ }
            }
            if (String.isBlank(correlationId)) {
                correlationId = 'NO-CORR-ID-' + DateTime.now().getTime();
            }

            // ── 2. Validate Content-Type ─────────────────────────────────────
            String contentType = req.headers.get('Content-Type');
            if (contentType == null || !contentType.containsIgnoreCase('application/json')) {
                sendResponse(res, 415, 'error', 'Content-Type must be application/json.',
                             correlationId);
                return;
            }

            // ── 3. Validate payload size ─────────────────────────────────────
            Integer bodyBytes = req.requestBody != null ? req.requestBody.size() : 0;
            if (bodyBytes > MAX_BODY_BYTES) {
                sendResponse(res, 413, 'error', 'Payload exceeds 64 KB limit.', correlationId);
                return;
            }

            // ── 4. Parse body ────────────────────────────────────────────────
            requestBody = rawBody;   // already extracted above
            if (String.isBlank(requestBody)) {
                sendResponse(res, 400, 'error', 'Request body is required.', correlationId);
                return;
            }

            Map<String, Object> body;
            try {
                body = (Map<String, Object>) JSON.deserializeUntyped(requestBody);
            } catch (JSONException je) {
                sendResponse(res, 400, 'error', 'Invalid JSON: ' + je.getMessage(),
                             correlationId);
                return;
            }

            // ── 5. Validate required envelope fields ─────────────────────────
            String eventType  = (String) body.get('Event_Type__c');
            String msgType    = (String) body.get('Msg_Type__c');
            String statusCode = (String) body.get('Status_Code__c');
            String clientId   = (String) body.get('Client_Id__c');
            String schemaName = (String) body.get('Schema_Name__c');
            String payload    = (String) body.get('Payload__c');
            String messageId  = (String) body.get('Message_Id__c');
            String userId     = (String) body.get('User_Id__c');
            String contentTypeFld = (String) body.get('Content_Type__c');

            if (String.isBlank(eventType) || String.isBlank(msgType) ||
                String.isBlank(statusCode) || String.isBlank(schemaName)) {
                sendResponse(res, 400, 'error',
                    'Missing required fields: Event_Type__c, Msg_Type__c, ' +
                    'Status_Code__c, Schema_Name__c.', correlationId);
                return;
            }

            // ── 6. Publish Platform Event ────────────────────────────────────
            AI_Navigator_Inbound__e evt = new AI_Navigator_Inbound__e(
                Message_Id__c     = String.isNotBlank(messageId)
                                    ? messageId
                                    : AI_Navigator_Event_Publisher.generateUUID(),
                Event_Type__c     = eventType,
                Msg_Type__c       = msgType,
                Status_Code__c    = statusCode,
                Client_Id__c      = clientId,
                User_Id__c        = userId,
                Correlation_Id__c = correlationId,
                Content_Type__c   = String.isNotBlank(contentTypeFld)
                                    ? contentTypeFld : 'json',
                Schema_Name__c    = schemaName,
                Payload__c        = payload,
                Created_At__c     = DateTime.now()
            );

            Database.SaveResult result = EventBus.publish(evt);

            if (!result.isSuccess()) {
                String errMsg = '';
                for (Database.Error err : result.getErrors()) {
                    errMsg += err.getMessage() + ' ';
                }
                sendResponse(res, 500, 'error',
                    'Failed to publish Platform Event.', correlationId);
                return;
            }

            // ── 7. Log and respond ───────────────────────────────────────────
            logRequest(correlationId, eventType, requestBody, 202);

            sendResponse(res, 202, 'accepted',
                'Event received and queued for processing.', correlationId);

        } catch (Exception e) {
            String errMsg = e.getTypeName() + ': ' + e.getMessage()
                          + ' | Line: ' + e.getLineNumber();
            System.debug(LoggingLevel.ERROR,
                'AzureInboundEventService: ' + errMsg);
            sendResponse(res, 500, 'error',
                'Internal Server Error. Contact the Salesforce admin.', correlationId);
        }
    }

    // ── Private: build and send JSON response ────────────────────────────────

    public static void publish(
        String eventType,
        String msgType,
        String schemaName,
        String payloadJson,

    ) {
        // Load config — fail fast if missing
        AI_Navigator_Config__mdt config = loadConfig();
        if (config == null) {

            return;
        }

            return;
        }

        // Validate payload size — Platform Events cap at 32,000 chars
        if (String.isNotBlank(payloadJson) && payloadJson.length() > 32000) {

                'for Event_Type: ' + eventType);
            return;
        }

        // Build the event

            Status_Code__c     = statusCode,
            Client_Id__c       = config.Client_Id__c,
            User_Id__c         = UserInfo.getUserId(),
            Correlation_Id__c  = generateUUID(),
            Content_Type__c    = 'json',
            Schema_Name__c     = schemaName + '.' + config.Event_Schema_Version__c,
            Payload__c         = payloadJson,
            Created_At__c      = DateTime.now()
        );

        // Publish
        Database.SaveResult result = EventBus.publish(evt);

        if (!result.isSuccess()) {
            for (Database.Error err : result.getErrors()) {

                    'AI_Navigator_Event_Publisher: Publish failed for ' + eventType +
                    ' | ' + err.getStatusCode() + ': ' + err.getMessage());
            }
        } else {

                'AI_Navigator_Event_Publisher: Published ' + eventType +
                ' | Correlation: ' + evt.Correlation_Id__c);

               hex.substring(16,20) + '-' +
               hex.substring(20,32);
    }

/**
 * AI_Navigator_Inbound_Handler
 *
 * Routes inbound Platform Events from Azure to the correct
 * processor based on Event_Type__c. All DML is bulkified —
 * the trigger may fire with multiple events at once.
 *
 * Supported Event Types:
 *   RESEARCH_DONE      → CompanyStatusUpdater
 *   UPLOAD_PROCESSED   → SubmissionProgressUpdater
 *   BATCH_PROGRESS     → SubmissionProgressUpdater
 *   SIGNAL_ALERT       → SignalAlertProcessor
 *   SCORING_DONE       → CompanyStatusUpdater
 *   REPORT_READY       → CompanyStatusUpdater
 *   PROCESSING_ERROR   → CompanyStatusUpdater
 */
public without sharing class AI_Navigator_Inbound_Handler {

    // ── Entry Point ──────────────────────────────────────────────────────────
    public static void handle(List<AI_Navigator_Inbound__e> events) {

        // Partition events by type for bulkified processing
        List<AI_Navigator_Inbound__e> researchEvents    = new List<AI_Navigator_Inbound__e>();
        List<AI_Navigator_Inbound__e> uploadEvents      = new List<AI_Navigator_Inbound__e>();
        List<AI_Navigator_Inbound__e> progressEvents    = new List<AI_Navigator_Inbound__e>();
        List<AI_Navigator_Inbound__e> signalEvents      = new List<AI_Navigator_Inbound__e>();
        List<AI_Navigator_Inbound__e> scoringEvents     = new List<AI_Navigator_Inbound__e>();
        List<AI_Navigator_Inbound__e> reportEvents      = new List<AI_Navigator_Inbound__e>();
        List<AI_Navigator_Inbound__e> errorEvents       = new List<AI_Navigator_Inbound__e>();

        for (AI_Navigator_Inbound__e evt : events) {

            // Log every inbound event to the request log
            logEvent(evt);

            if (String.isBlank(evt.Event_Type__c)) {
                System.debug(LoggingLevel.WARN,
                    'AI_Navigator_Inbound_Handler: Received event with no Event_Type__c. ' +
                    'Message_Id: ' + evt.Message_Id__c);
                continue;
            }

            switch on evt.Event_Type__c {
                when 'RESEARCH_DONE'    { researchEvents.add(evt); }
                when 'UPLOAD_PROCESSED' { uploadEvents.add(evt); }
                when 'BATCH_PROGRESS'   { progressEvents.add(evt); }
                when 'SIGNAL_ALERT'     { signalEvents.add(evt); }
                when 'SCORING_DONE'     { scoringEvents.add(evt); }
                when 'REPORT_READY'     { reportEvents.add(evt); }
                when 'PROCESSING_ERROR' { errorEvents.add(evt); }
                when else {
                    System.debug(LoggingLevel.WARN,
                        'AI_Navigator_Inbound_Handler: Unknown Event_Type__c: ' +
                        evt.Event_Type__c + ' | Message_Id: ' + evt.Message_Id__c);
                }
            }
        }

        // ── Dispatch to processors ───────────────────────────────────────────
        if (!researchEvents.isEmpty()) {
            CompanyStatusUpdater.processResearchDone(researchEvents);
        }
        if (!uploadEvents.isEmpty()) {
            SubmissionProgressUpdater.processUploadComplete(uploadEvents);
        }
        if (!progressEvents.isEmpty()) {
            SubmissionProgressUpdater.processBatchProgress(progressEvents);
        }
        if (!signalEvents.isEmpty()) {
            SignalAlertProcessor.processSignalAlerts(signalEvents);
        }
        if (!scoringEvents.isEmpty()) {
            CompanyStatusUpdater.processScoringDone(scoringEvents);
        }
        if (!reportEvents.isEmpty()) {
            CompanyStatusUpdater.processReportReady(reportEvents);
        }
        if (!errorEvents.isEmpty()) {
            CompanyStatusUpdater.processError(errorEvents);
        }
    }

    // ── Private: log every event to Azure_Request_Log__c ────────────────────
    private static void logEvent(AI_Navigator_Inbound__e evt) {
        try {
            Azure_Request_Log__c log = new Azure_Request_Log__c(
                HTTP_Method__c      = 'PLATFORM_EVENT',
                Endpoint_Name__c    = 'AI_Navigator_Inbound__e',
                Request_Body__c     = evt.Payload__c,
                Status__c           = evt.Status_Code__c,
                Correlation_ID__c   = evt.Correlation_Id__c,
                Error_Message__c    = evt.Event_Type__c + ' | ' + evt.Schema_Name__c
            );
            insert log;
        } catch (DmlException de) {
            System.debug(LoggingLevel.ERROR,
                'AI_Navigator_Inbound_Handler: Failed to write log. ' + de.getMessage());
        }
    }
}

 * from a Platform Event trigger context.

    public static void handle(List<AI_Navigator_Outbound__e> events) {

                + ' | correlationId=' + correlationId + ' | error=' + errMsg);

                'Failed to send request to processing pipeline. Please try again. ' + errMsg
            );

            'AiNavigatorEventService: Published ' + eventType
            + ' | schema=' + schemaName + ' | correlationId=' + correlationId);

     * @description Generates a Salesforce-owned correlation ID in the format
     *              Azure expects for end-to-end trace correlation.
     *
     *   Format  : SF-{unix-ms}-{8-char-lowercase-hex}
     *   Example : SF-1714060800000-a3f9d2c1
     *
     *   The 8-char hex tail provides ~4 billion values per millisecond bucket —
     *   sufficient entropy for trace correlation per Azure spec v2.
     *
     *   CHANGE FROM PREVIOUS VERSION:
     *     Old: 4-char uppercase hex  → SF-1745271600000-A3F2
     *     New: 8-char lowercase hex  → SF-1714060800000-a3f9d2c1
     *     Reason: Azure developer confirmed format requirement 2026-04-27.
     */
    public static String generateCorrelationId() {
        String ts   = String.valueOf(DateTime.now().getTime());
        String rand = EncodingUtil.convertToHex(
                          Crypto.generateAesKey(128)
                      ).substring(0, 8).toLowerCase();
        return 'SF-' + ts + '-' + rand;
    }

/**
 * AzureCompanyCallbackService
 *
 * Direct HTTP callback endpoint. Azure calls this to update a
 * specific Company__c record with research results, scores, or
 * status changes without going through the Platform Event bus.
 * Used for lightweight updates where full event routing is overkill.
 *
 * Endpoint : /services/apexrest/ainavigator/v1/companies/callback
 * Method   : POST
 * Auth     : OAuth 2.0 Bearer Token (Client Credentials Flow)
 *
 * Expected body:
 * {
 *   "companyId"    : "a0Bxx000001SvgsEAC",   // Salesforce record ID
 *   "status"       : "completed",             // pending/processing/completed/failed
 *   "prospectScore": 78.5,                    // optional
 *   "pdfReportUrl" : "https://...",           // optional
 *   "correlationId": "uuid"                   // optional
 * }
 */
@RestResource(urlMapping='/ainavigator/v1/companies/callback')
global without sharing class AzureCompanyCallbackService {

    private static final String CORR_HEADER = 'X-Correlation-Id';

    @HttpPost
    global static void handlePost() {

        RestRequest  req = RestContext.request;
        RestResponse res = RestContext.response;

        String correlationId = '';
        String requestBody   = '';

        try {
            // ── 1. Headers ───────────────────────────────────────────────────
            correlationId = req.headers.get(CORR_HEADER);
            if (String.isBlank(correlationId)) {
                correlationId = 'NO-CORR-ID-' + DateTime.now().getTime();
            }

            // ── 2. Parse body ────────────────────────────────────────────────
            requestBody = req.requestBody != null ? req.requestBody.toString() : '';
            if (String.isBlank(requestBody)) {
                sendResponse(res, 400, 'error', 'Request body is required.', correlationId);
                return;
            }

            Map<String, Object> body;
            try {
                body = (Map<String, Object>) JSON.deserializeUntyped(requestBody);
            } catch (JSONException je) {
                sendResponse(res, 400, 'error', 'Invalid JSON.', correlationId);
                return;
            }

            // ── 3. Validate companyId ────────────────────────────────────────
            String companyId = (String) body.get('companyId');
            if (String.isBlank(companyId)) {
                sendResponse(res, 400, 'error', 'companyId is required.', correlationId);
                return;
            }

            // ── 4. Fetch company ─────────────────────────────────────────────
            List<Company__c> companies = [
                SELECT Id, Status__c, Prospect_Score__c, Scored_At__c,
                       PDF_Report_URL__c, Research_Completed_At__c,
                       Correlation_Id__c
                FROM Company__c
                WHERE Id = :companyId
                LIMIT 1
            ];

            if (companies.isEmpty()) {
                sendResponse(res, 404, 'error',
                    'Company not found: ' + companyId, correlationId);
                return;
            }

            Company__c company = companies[0];

            // ── 5. Apply updates from payload ────────────────────────────────
            String newStatus = (String) body.get('status');
            if (String.isNotBlank(newStatus)) {
                company.Status__c = newStatus;
                if (newStatus == 'completed' || newStatus == 'failed') {
                    company.Research_Completed_At__c = DateTime.now();
                }
            }

            Object scoreVal = body.get('prospectScore');
            if (scoreVal != null) {
                company.Prospect_Score__c = Decimal.valueOf(String.valueOf(scoreVal));
                company.Scored_At__c      = DateTime.now();
            }

            String pdfUrl = (String) body.get('pdfReportUrl');
            if (String.isNotBlank(pdfUrl)) {
                company.PDF_Report_URL__c = pdfUrl;
            }

            company.Correlation_Id__c = correlationId;

            // ── 6. Save ──────────────────────────────────────────────────────
            try {
                update company;
            } catch (DmlException de) {
                sendResponse(res, 500, 'error',
                    'Failed to update company record.', correlationId);

                              correlationId, requestBody, sourceIp, bodyBytes);
                    return;
                }

            Map<String, Object> echoPayload = new Map<String, Object>{
                'status'        => 'success',
                'correlationId' => correlationId,
                'echo'          => String.isNotBlank(requestBody)

                                   : new Map<String, Object>(),
                'receivedAt'    => String.valueOf(DateTime.now()),
                'endpoint'      => ENDPOINT_NAME
            };

/**
 * @description Shared HTTP utility for all outbound Azure APIM callouts.
 *              Builds pre-configured HttpRequest objects using headers pulled
 *              from AI_Navigator_Config__mdt (DeveloperName = 'Dev').
 *
 *              Required Azure headers:
 *                - Ocp-Apim-Subscription-Key  ← new ApimSubscriptionKey__c field
 *                - X-Client-Id                ← Client_Id__c from CMT
 *                - X-User-Id                  ← SF User Id of submitting user
 *                - X-Correlation-Id           ← UUID generated per request
 *                - Content-Type application/json
 *                - Authorization Bearer       ← injected by Named Credential
 */
public without sharing class AzureHttpService {

    private static final Integer DEFAULT_TIMEOUT_MS = 30000;

    /**
     * Build a ready-to-send HttpRequest with all required Azure headers.
     *
     * @param method           HTTP verb: POST, GET, PUT
     * @param path             Path appended to Named Credential base URL
     *                         e.g. /api/v1/user/companies/manual
     * @param correlationId    UUID generated by the caller
     * @param submittingUserId Salesforce User Id of the user who triggered the action
     * @return Fully configured HttpRequest — caller sets body if needed, then sends
     */
    public static HttpRequest buildRequest(
        String method,
        String path,
        String correlationId,
        String submittingUserId
    ) {
        AI_Navigator_Config__mdt config = loadConfig();

        HttpRequest req = new HttpRequest();

        // Named Credential handles Authorization: Bearer <JWT> automatically
        req.setEndpoint('callout:' + config.Named_Credential_Name__c + path);
        req.setMethod(method);
        req.setTimeout(
            config.Callout_Timeout_Ms__c != null
                ? Integer.valueOf(config.Callout_Timeout_Ms__c)
                : DEFAULT_TIMEOUT_MS
        );

        // Required headers — matched to Azure APIM contract
        req.setHeader('Content-Type',               'application/json');
        req.setHeader('Accept',                     'application/json');
        req.setHeader('Ocp-Apim-Subscription-Key',  config.ApimSubscriptionKey__c);
        req.setHeader('X-Client-Id',                config.Client_Id__c);
        req.setHeader('X-User-Id',
            String.isNotBlank(submittingUserId) ? submittingUserId : UserInfo.getUserId()
        );
        req.setHeader('X-Correlation-Id',           correlationId);

        return req;
    }

    /**
     * Parse a standard Azure error response envelope and return a readable message.
     */
    public static String parseErrorResponse(HttpResponse res) {
        try {
            Map<String, Object> body =
                (Map<String, Object>) JSON.deserializeUntyped(res.getBody());
            String msg    = (String) body.get('message');
            String code   = (String) body.get('code');
            String corrId = (String) body.get('correlationId');
            return (code   != null ? '[' + code   + '] ' : '')
                 + (msg    != null ? msg            : res.getBody())
                 + (corrId != null ? ' (correlationId: ' + corrId + ')' : '');
        } catch (Exception e) {
            return 'HTTP ' + res.getStatusCode() + ' — ' + res.getBody();
        }
    }

    // ── Private ──────────────────────────────────────────────────────────────

    private static AI_Navigator_Config__mdt loadConfig() {
        List<AI_Navigator_Config__mdt> configs = [
            SELECT Named_Credential_Name__c,
                   ApimSubscriptionKey__c,
                   Client_Id__c,
                   Callout_Timeout_Ms__c,
                   Is_Active__c
            FROM   AI_Navigator_Config__mdt
            WHERE  DeveloperName = 'Dev'
            LIMIT  1
        ];

        if (configs.isEmpty() || configs[0].Named_Credential_Name__c == null) {
            throw new AzureHttpServiceException(
                'AI_Navigator_Config__mdt "Dev" record is missing or incomplete.'
            );
        }
        if (!configs[0].Is_Active__c) {
            throw new AzureHttpServiceException(
                'Azure callouts are disabled via kill switch (Is_Active__c = false).'
            );
        }
        return configs[0];
    }

    public class AzureHttpServiceException extends Exception {}
}

    private static final String CLIENT_HDR       = 'X-Client-Id';

        RestResponse res = RestContext.response;

        String correlationId = '';
        String requestBody   = '';

            // ── 1. Extract headers ───────────────────────────────────────────
            // Headers are case-sensitive in RestContext — try both casings,
            // then fall back to the body field Correlation_Id__c.
            correlationId = req.headers.get('X-Correlation-Id');
            if (String.isBlank(correlationId)) {
                correlationId = req.headers.get('x-correlation-id');

            if (String.isBlank(correlationId)) {
                correlationId = req.headers.get('X-CORRELATION-ID');
            }
            // Body field fallback — Azure also sends it as Correlation_Id__c
            // Parse body first so we can read it here
            String rawBody = req.requestBody != null ? req.requestBody.toString() : '';
            if (String.isBlank(correlationId) && String.isNotBlank(rawBody)) {
                try {
                    Map<String, Object> tempBody =
                        (Map<String, Object>) JSON.deserializeUntyped(rawBody);
                    correlationId = (String) tempBody.get('Correlation_Id__c');
                } catch (Exception e) { /* ignore — full parse happens in step 4 */ }
            }
            if (String.isBlank(correlationId)) {
                correlationId = 'NO-CORR-ID-' + DateTime.now().getTime();
            }

            // ── 2. Validate Content-Type ─────────────────────────────────────
            String contentType = req.headers.get('Content-Type');
            if (contentType == null || !contentType.containsIgnoreCase('application/json')) {
                sendResponse(res, 415, 'error', 'Content-Type must be application/json.',
                             correlationId);
                return;
            }

            // ── 3. Validate payload size ─────────────────────────────────────
            Integer bodyBytes = req.requestBody != null ? req.requestBody.size() : 0;
            if (bodyBytes > MAX_BODY_BYTES) {
                sendResponse(res, 413, 'error', 'Payload exceeds 64 KB limit.', correlationId);
                return;
            }

            // ── 4. Parse body ────────────────────────────────────────────────
            requestBody = rawBody;   // already extracted above
            if (String.isBlank(requestBody)) {
                sendResponse(res, 400, 'error', 'Request body is required.', correlationId);

            Map<String, Object> body;

            String statusCode = (String) body.get('Status_Code__c');

            String userId     = (String) body.get('User_Id__c');

/**

 * Usage:
 *   AzureSignalService.requestSignalAnalysis(companyId, recencyFilter);
 */
public without sharing class AzureSignalService {

    public static void requestSignalAnalysis(String companyId, String recencyFilter) {
        List<Company__c> companies = [
            SELECT Id, Name, Website__c, Industry__c, Client_Type__c
            FROM Company__c
            WHERE Id = :companyId
            LIMIT 1
        ];

        if (companies.isEmpty()) {
            System.debug(LoggingLevel.ERROR,
                'AzureSignalService: Company not found: ' + companyId);
            return;
        }

        Company__c company = companies[0];

        // Default recency filter to month if not provided
        String filter = String.isBlank(recencyFilter) ? 'month' : recencyFilter;

        // Build payload matching SignalAnalysisRequest.v1 schema
        Map<String, Object> payload = new Map<String, Object>{
            'companyName'        => company.Name,
            'companyUrl'         => company.Website__c,
            'industry'           => company.Industry__c,
            'relationshipStatus' => mapClientTypeToRelationship(company.Client_Type__c),
            'recencyFilter'      => filter
        };

        AI_Navigator_Event_Publisher.publish(
            'SIGNAL_ANALYSIS',
            'Request',
            'SignalAnalysisRequest',
            JSON.serialize(payload)
        );
    }

    // Map Client_Type__c value to Azure relationship status enum
    private static String mapClientTypeToRelationship(String clientType) {
        if (String.isBlank(clientType)) return 'prospect';
        String ct = clientType.toLowerCase();
        if (ct.contains('existing') || ct.contains('customer')) return 'existing_customer';
        if (ct.contains('past'))                                 return 'past_customer';
        return 'prospect';
    }
}

/**
 * AzureUploadCompanyCallout
 * Queueable + AllowsCallouts — fires the HTTP POST to Azure after
 * UploadController creates the Submission__c and Company__c records.
 *
 * Request body matches UploadRequest.v1 schema:
 * { fileName, fileContent, submissionId, companies[], stage, clientType,
 *   fileSize, submittedBy, submittedAt, correlationId, schemaVersion }
 * correlationId is ALSO sent as header X-Correlation-Id via AzureHttpService.
 */
public class AzureUploadCompanyCallout implements Queueable, Database.AllowsCallouts {

    private String submissionRecordId;
    private String payload;
    private String correlationId;

    public AzureUploadCompanyCallout(
        String submissionRecordId,
        String payload,
        String correlationId
    ) {
        this.submissionRecordId = submissionRecordId;
        this.payload            = payload;
        this.correlationId      = correlationId;
    }

    public void execute(QueueableContext context) {

        HttpRequest req = AzureHttpService.buildRequest(
            'POST',
            '/v1/upload',              // ← confirm with Azure team
            correlationId,
            UserInfo.getUserId()
        );
        req.setTimeout(120000);        // ← override default — fileContent is large
        req.setBody(payload);

        HttpResponse res = new Http().send(req);

        // ── LOG THE OUTBOUND REQUEST ─────────────────────────────────────────
        try {
            insert new Azure_Request_Log__c(
                HTTP_Method__c      = 'POST',
                Endpoint_Name__c    = 'AzureUploadCompanyCallout',
                Response_Body__c    = res.getBody(),
                Status__c           = (res.getStatusCode() == 200 || res.getStatusCode() == 202)
                                        ? 'Success' : 'Failed',
                HTTP_Status_Code__c = res.getStatusCode(),
                Correlation_ID__c   = correlationId,
                Error_Message__c    = (res.getStatusCode() == 200 || res.getStatusCode() == 202)
                                        ? 'submissionId=' + submissionRecordId
                                        : AzureHttpService.parseErrorResponse(res)
            );
        } catch (Exception logEx) {
            System.debug(LoggingLevel.ERROR, 'Log write failed: ' + logEx.getMessage());
        }
        // ────────────────────────────────────────────────────────────────────

        if (res.getStatusCode() == 200 || res.getStatusCode() == 202) {
            update new Submission__c(
                Id        = submissionRecordId,
                Status__c = 'submitted'
            );
        } else {
            update new Submission__c(
                Id        = submissionRecordId,
                Status__c = 'failed'
            );
            System.debug(LoggingLevel.ERROR,
                'AzureUploadCompanyCallout failed ['
                + res.getStatusCode() + ']: '
                + AzureHttpService.parseErrorResponse(res));
        }
    }
}

    public static void submitUpload(
        String submissionId,
        String fileName,
        String blobPath,
        String stage,
        String clientType
    ) {
        // Create the Submission__c tracking record
        Submission__c sub = new Submission__c(
            Submission_Id__c   = submissionId,
            File_Name__c       = fileName,
            Blob_Path__c       = blobPath,
            Status__c          = 'pending',
            Submitted_At__c    = DateTime.now(),
            Submitted_By__c    = UserInfo.getUserId()
        );

            insert sub;

        // Build payload matching UploadRequest.v1 schema

            'submissionId' => submissionId,
            'fileName'     => fileName,
            'blobPath'     => blobPath,
            'stage'        => stage,
            'clientType'   => clientType
        };

        AI_Navigator_Event_Publisher.publish(
            'UPLOAD',
            'Request',

/**
 * @description Handles inbound Platform Events that update Company__c records.
 *
 *   Supported event types:
 *     - RESEARCH_DONE         → Status__c = completed/failed, Research_Completed_At__c
 *     - SCORING_DONE          → Prospect_Score__c, Scored_At__c
 *     - REPORT_READY          → PDF_Report_URL__c  (or report_failed if code indicates failure)
 *     - PROCESSING_ERROR      → Status__c = failed, Error_Message__c
 *
 *   LOOKUP STRATEGY:
 *     All handlers match on Correlation_Id__c — a dedicated field on the Platform
 *     Event that Azure echoes back from the original upload request.
 *     It is stored on Company__c at record-creation time, enabling a direct
 *     WHERE Correlation_Id__c IN lookup with no Azure changes required.
 *
 * @group AI Navigator Integration
 * @last-modified 2026-05-04
 */
public without sharing class CompanyStatusUpdater {

    // ─────────────────────────────────────────────────────────────────────────
    // RESEARCH_DONE
    // ─────────────────────────────────────────────────────────────────────────
    public static void processResearchDone(List<AI_Navigator_Inbound__e> events) {
        Map<String, AI_Navigator_Inbound__e> corrIdToEvent = mapByCorrelationId(events);
        if (corrIdToEvent.isEmpty()) return;

        List<Company__c> toUpdate = new List<Company__c>();
        for (Company__c company : [
            SELECT Id, Status__c, Research_Completed_At__c,
                   Correlation_Id__c, Azure_Company_Id__c, Error_Message__c
            FROM   Company__c
            WHERE  Correlation_Id__c IN :corrIdToEvent.keySet()
        ]) {
            AI_Navigator_Inbound__e evt     = corrIdToEvent.get(company.Correlation_Id__c);
            Map<String, Object>     payload = parsePayload(evt.Payload__c);

            String status = (String) payload.get('status');
            if (status == 'failed') {
                company.Status__c        = 'failed';
                company.Error_Message__c = extractErrorMessage(payload);
            } else {
                company.Status__c                = 'completed';
                company.Research_Completed_At__c = DateTime.now();
                company.Error_Message__c         = null; // clear any prior error
            }
            company.Azure_Company_Id__c = (String) payload.get('companyId');
            toUpdate.add(company);
        }
        updateRecords(toUpdate, 'CompanyStatusUpdater.processResearchDone');
    }

    // ─────────────────────────────────────────────────────────────────────────
    // SCORING_DONE
    // ─────────────────────────────────────────────────────────────────────────
    public static void processScoringDone(List<AI_Navigator_Inbound__e> events) {
        Map<String, AI_Navigator_Inbound__e> corrIdToEvent = mapByCorrelationId(events);
        if (corrIdToEvent.isEmpty()) return;

        List<Company__c> toUpdate = new List<Company__c>();
        for (Company__c company : [
            SELECT Id, Prospect_Score__c, Scored_At__c,
                   Correlation_Id__c, Azure_Company_Id__c, Error_Message__c
            FROM   Company__c
            WHERE  Correlation_Id__c IN :corrIdToEvent.keySet()
        ]) {
            AI_Navigator_Inbound__e evt     = corrIdToEvent.get(company.Correlation_Id__c);
            Map<String, Object>     payload = parsePayload(evt.Payload__c);

            Object scoreVal = payload.get('overallScore');
            if (scoreVal != null) {
                company.Prospect_Score__c = Decimal.valueOf(String.valueOf(scoreVal));
                company.Scored_At__c      = DateTime.now();
                company.Error_Message__c  = null;
            } else {
                company.Error_Message__c = extractErrorMessage(payload);
            }
            company.Azure_Company_Id__c = (String) payload.get('companyId');
            toUpdate.add(company);
        }
        updateRecords(toUpdate, 'CompanyStatusUpdater.processScoringDone');
    }

    // ─────────────────────────────────────────────────────────────────────────
    // REPORT_READY  (also handles AI_NAVIGATOR_REPORT_FAILED inside this event)
    // ─────────────────────────────────────────────────────────────────────────
    public static void processReportReady(List<AI_Navigator_Inbound__e> events) {
        Map<String, AI_Navigator_Inbound__e> corrIdToEvent = mapByCorrelationId(events);
        if (corrIdToEvent.isEmpty()) return;

        List<Company__c> toUpdate = new List<Company__c>();
        for (Company__c company : [
            SELECT Id, Status__c, PDF_Report_URL__c,
                   Correlation_Id__c, Azure_Company_Id__c, Error_Message__c
            FROM   Company__c
            WHERE  Correlation_Id__c IN :corrIdToEvent.keySet()
        ]) {
            AI_Navigator_Inbound__e evt     = corrIdToEvent.get(company.Correlation_Id__c);
            Map<String, Object>     payload = parsePayload(evt.Payload__c);
            String                  code    = (String) payload.get('code');

            // Azure sometimes sends failure inside a REPORT_READY event
            if (code != null && code.containsIgnoreCase('FAILED')) {
                company.Status__c        = 'report_failed';
                company.Error_Message__c = extractErrorMessage(payload);
            } else if (evt.Content_Type__c == 'uri') {
                company.PDF_Report_URL__c = evt.Payload__c;
                company.Error_Message__c  = null;
            } else {
                company.PDF_Report_URL__c = (String) payload.get('downloadUrl');
                company.Error_Message__c  = null;
            }
            company.Azure_Company_Id__c = (String) payload.get('companyId');
            toUpdate.add(company);
        }
        updateRecords(toUpdate, 'CompanyStatusUpdater.processReportReady');
    }

    // ─────────────────────────────────────────────────────────────────────────
    // PROCESSING_ERROR
    // ─────────────────────────────────────────────────────────────────────────
    public static void processError(List<AI_Navigator_Inbound__e> events) {
        Map<String, AI_Navigator_Inbound__e> corrIdToEvent = mapByCorrelationId(events);
        if (corrIdToEvent.isEmpty()) return;

        List<Company__c> toUpdate = new List<Company__c>();
        for (Company__c company : [
            SELECT Id, Status__c,
                   Correlation_Id__c, Azure_Company_Id__c, Error_Message__c
            FROM   Company__c
            WHERE  Correlation_Id__c IN :corrIdToEvent.keySet()
        ]) {
            AI_Navigator_Inbound__e evt     = corrIdToEvent.get(company.Correlation_Id__c);
            Map<String, Object>     payload = parsePayload(evt.Payload__c);

            company.Status__c           = 'failed';
            company.Error_Message__c    = extractErrorMessage(payload);
            company.Azure_Company_Id__c = (String) payload.get('companyId');
            toUpdate.add(company);
        }
        updateRecords(toUpdate, 'CompanyStatusUpdater.processError');
    }

    // ─────────────────────────────────────────────────────────────────────────
    // Private Helpers
    // ─────────────────────────────────────────────────────────────────────────

    private static Map<String, AI_Navigator_Inbound__e> mapByCorrelationId(
        List<AI_Navigator_Inbound__e> events
    ) {
        Map<String, AI_Navigator_Inbound__e> result = new Map<String, AI_Navigator_Inbound__e>();
        for (AI_Navigator_Inbound__e evt : events) {
            if (String.isNotBlank(evt.Correlation_Id__c)) {
                result.put(evt.Correlation_Id__c, evt);
            }
        }
        return result;
    }

    /**
     * @description Extracts a clean, user-friendly error message from the payload.
     *              Prefers 'message' field, falls back to 'code', then a generic string.
     *              Truncates to 255 chars to fit a standard text field.
     */
    private static String extractErrorMessage(Map<String, Object> payload) {
        String msg  = (String) payload.get('message');
        String code = (String) payload.get('code');
        String raw  = String.isNotBlank(msg) ? msg
                    : String.isNotBlank(code) ? code
                    : 'Azure processing error — no details provided';
        // Strip HTML tags Azure sometimes includes in error messages
        raw = raw.replaceAll('<[^>]+>', '').trim();
        return raw.length() > 255 ? raw.substring(0, 255) : raw;
    }

    private static Map<String, Object> parsePayload(String payloadJson) {
        if (String.isBlank(payloadJson)) return new Map<String, Object>();
        try {
            return (Map<String, Object>) JSON.deserializeUntyped(payloadJson);
        } catch (Exception e) {
            System.debug(LoggingLevel.ERROR,
                'CompanyStatusUpdater: Failed to parse payload. ' + e.getMessage());
            return new Map<String, Object>();
        }
    }

    private static void updateRecords(List<Company__c> records, String context) {
        if (records.isEmpty()) return;
        try {
            update records;
        } catch (DmlException de) {
            System.debug(LoggingLevel.ERROR,
                context + ': DML failed. ' + de.getMessage());
        }
    }

    private static HttpResponse azureCallout(
        String method, String path,
        Map<String, Object> body, Integer timeoutMs
    ) {
        HttpRequest req = new HttpRequest();
        req.setEndpoint(NAMED_CRED + path);
        req.setMethod(method);
        req.setTimeout(timeoutMs);
        req.setHeader('Content-Type', 'application/json');
        req.setHeader('X-Correlation-Id', generateCorrelationId());
        if (body != null && !body.isEmpty()) {
            req.setBody(JSON.serialize(body));
        }
        return new Http().send(req);
    }

    @TestVisible
    private static void handleCalloutResponse(HttpResponse res, String operation, String azureId) {
        Integer statusCode = res.getStatusCode();
        if (statusCode >= 200 && statusCode < 300) { return; }

        String correlationId = res.getHeader('X-Correlation-Id');
        String errorMsg;

        if      (statusCode == 400) { errorMsg = operation + ': Validation error (400). Azure Id: ' + azureId; }
        else if (statusCode == 401) { errorMsg = operation + ': Authentication failed (401). Check Named Credential token.'; }
        else if (statusCode == 403) { errorMsg = operation + ': Forbidden (403).'; }
        else if (statusCode == 404) { errorMsg = operation + ': Company not found on Azure (404). Azure Id: ' + azureId; }
        else if (statusCode == 409) { errorMsg = operation + ': Conflict (409). ' + res.getBody(); }
        else if (statusCode == 429) { errorMsg = operation + ': Rate limited (429). Retry-After: ' + res.getHeader('Retry-After'); }
        else if (statusCode == 500) { errorMsg = operation + ': Azure internal error (500). CorrelationId: ' + correlationId; }
        else if (statusCode == 502 || statusCode == 503) { errorMsg = operation + ': Upstream unavailable (' + statusCode + ').'; }
        else if (statusCode == 504) { errorMsg = operation + ': Gateway timeout (504). Verify state before retrying.'; }
        else                        { errorMsg = operation + ': Unexpected status (' + statusCode + '). CorrelationId: ' + correlationId; }

        System.debug(LoggingLevel.ERROR, 'DashboardController.' + errorMsg);

        try {
             insert new Azure_Request_Log__c(
                HTTP_Method__c      = operation,
                Endpoint_Name__c    = operation,
                Status__c           = 'Error',
                HTTP_Status_Code__c = statusCode,
                Correlation_ID__c   = correlationId,
                Error_Message__c    = errorMsg
            );
        } catch (DmlException de) {
            System.debug(LoggingLevel.WARN, 'DashboardController: failed to write request log: ' + de.getMessage());
        }

        if (statusCode != 404) {
            throw new AuraHandledException(errorMsg);
        }
    }

    private static String generateCorrelationId() {
        return EncodingUtil.convertToHex(Crypto.generateAesKey(128)).substring(0, 8)
            + '-' + String.valueOf(DateTime.now().getTime());
    }

/**
 * @description Controller for the inputForm LWC.
 *              1. Creates Company__c with Status__c = 'processing'
 *              2. Returns SF record ID to LWC immediately (no waiting on Azure)
 *              3. Enqueues AzureManualCompanyCallout — fires async after commit
 */
public with sharing class InputController {

    @AuraEnabled
    public static String submitCompany(
        String companyName,
        String website,
        String city,
        String state,
        String stage,
        String clientType
    ) {
        try {
            // 1. Generate correlation ID
            String correlationId = AI_Navigator_Event_Publisher.generateCorrelationId();
            String currentUserId = UserInfo.getUserId();

            // 2. Create Company__c — status pending until Azure confirms
            Company__c company = new Company__c(
                Name             = companyName,
                Website__c       = website,
                City__c          = city,
                State__c         = state,
                Stage__c         = stage,
                Client_Type__c   = clientType,
                Status__c        = 'pending',
                Correlation_Id__c = correlationId
            );
            insert company;

            // 3. Enqueue async callout — runs after DML commits
            //    Carries all fields needed to build the Azure payload
            System.enqueueJob(new AzureManualCompanyCallout(
                                company.Id,      // recordId — used to update the Company__c after callout
                                companyName,
                                website,
                                city,
                                state,
                                stage,
                                clientType,
                                company.Id       // salesforceId — echoed back by Azure on every callback
                            ));

            // 4. Return SF record data to LWC immediately
            return JSON.serialize(new Map<String, Object>{
                'success'       => true,
                'companyId'     => company.Id,
                'correlationId' => correlationId,
                'name'          => companyName,
                'website'       => website,
                'city'          => city,
                'state'         => state,
                'stage'         => stage,
                'clientType'    => clientType,
                'status'        => 'pending',
                'createdAt'     => String.valueOf(DateTime.now())
            });

        } catch (DmlException dmlEx) {
            System.debug(LoggingLevel.ERROR,
                'InputController.submitCompany DML error: ' + dmlEx.getMessage());
            throw new AuraHandledException(
                'Could not save the company record: ' + dmlEx.getDmlMessage(0)
            );
        } catch (AuraHandledException ahe) {
            throw ahe;
        } catch (Exception e) {
            System.debug(LoggingLevel.ERROR,
                'InputController.submitCompany unexpected: ' + e.getMessage());
            throw new AuraHandledException(
                'An unexpected error occurred. Please try again.'
            );
        }
    }
}

/**
 * SignalAlertProcessor
 *
 * Handles SIGNAL_ALERT inbound events from Azure.
 * Creates Signal__c child records under the matching Company__c.
 * Bulkified — processes multiple events and multiple signals per event.
 */
public without sharing class SignalAlertProcessor {

    public static void processSignalAlerts(List<AI_Navigator_Inbound__e> events) {

        // Step 1 — collect all companyIds from payloads
        Map<String, List<Map<String, Object>>> companyIdToSignals =
            new Map<String, List<Map<String, Object>>>();

        for (AI_Navigator_Inbound__e evt : events) {
            Map<String, Object> payload = parsePayload(evt.Payload__c);
            String companyId = (String) payload.get('companyId');
            if (String.isBlank(companyId)) continue;

            List<Object> rawSignals = (List<Object>) payload.get('signals');
            if (rawSignals == null || rawSignals.isEmpty()) continue;

            List<Map<String, Object>> signalMaps = new List<Map<String, Object>>();
            for (Object s : rawSignals) {
                signalMaps.add((Map<String, Object>) s);
            }

            if (!companyIdToSignals.containsKey(companyId)) {
                companyIdToSignals.put(companyId, new List<Map<String, Object>>());
            }
            companyIdToSignals.get(companyId).addAll(signalMaps);
        }

        if (companyIdToSignals.isEmpty()) return;

        // Step 2 — verify companies exist
        Set<Id> validCompanyIds = new Set<Id>();
        for (Company__c c : [
            SELECT Id FROM Company__c
            WHERE Id IN :companyIdToSignals.keySet()
        ]) {
            validCompanyIds.add(c.Id);
        }

        // Step 3 — build Signal__c records
        List<Signal__c> toInsert = new List<Signal__c>();

        for (String companyId : companyIdToSignals.keySet()) {
            if (!validCompanyIds.contains((Id) companyId)) continue;

            for (Map<String, Object> s : companyIdToSignals.get(companyId)) {
                toInsert.add(new Signal__c(
                    Company__c    = (Id) companyId,
                    Signal_Type__c = (String)  s.get('signalType'),
                    Headline__c   = (String)  s.get('headline'),
                    Impact__c     = (String)  s.get('impact'),
                    Priority__c   = (String)  s.get('priority'),
                    Source__c     = (String)  s.get('source'),
                    Detected_At__c = parseDateTime(s.get('detectedAt')),
                    Is_New__c     = true
                ));
            }
        }

        if (toInsert.isEmpty()) return;

        try {
            insert toInsert;
        } catch (DmlException de) {
            System.debug(LoggingLevel.ERROR,
                'SignalAlertProcessor: DML failed. ' + de.getMessage());
        }
    }

    // ── Private Helpers ──────────────────────────────────────────────────────
    private static Map<String, Object> parsePayload(String payloadJson) {
        if (String.isBlank(payloadJson)) return new Map<String, Object>();
        try {
            return (Map<String, Object>) JSON.deserializeUntyped(payloadJson);
        } catch (Exception e) {
            System.debug(LoggingLevel.ERROR,
                'SignalAlertProcessor: Failed to parse payload. ' + e.getMessage());
            return new Map<String, Object>();
        }
    }

    private static DateTime parseDateTime(Object val) {
        if (val == null) return DateTime.now();
        try {
            return (DateTime) JSON.deserialize('"' + String.valueOf(val) + '"', DateTime.class);
        } catch (Exception e) {
            return DateTime.now();
        }
    }
}

/**
 * SubmissionProgressUpdater
 *
 * Handles inbound Platform Events that update Submission__c records:
 *   - UPLOAD_PROCESSED → Final status, completed/partial/failed counts
 *   - BATCH_PROGRESS   → Interim processed count for progress bar
 */
public without sharing class SubmissionProgressUpdater {

    // ── UPLOAD_PROCESSED ─────────────────────────────────────────────────────
    public static void processUploadComplete(List<AI_Navigator_Inbound__e> events) {
        Map<String, AI_Navigator_Inbound__e> subIdToEvent = mapBySubmissionId(events);
        if (subIdToEvent.isEmpty()) return;

        List<Submission__c> toUpdate = new List<Submission__c>();

        for (Submission__c sub : [
            SELECT Id, Submission_Id__c, Status__c,
                   Total_Companies__c, Processed_Companies__c,
                   Failed_Companies__c, Completed_At__c
            FROM Submission__c
            WHERE Submission_Id__c IN :subIdToEvent.keySet()
        ]) {
            AI_Navigator_Inbound__e evt = subIdToEvent.get(sub.Submission_Id__c);
            Map<String, Object> payload = parsePayload(evt.Payload__c);

            Integer total     = intVal(payload.get('totalCompanies'));
            Integer completed = intVal(payload.get('completed'));
            Integer failed    = intVal(payload.get('failed'));

            sub.Total_Companies__c     = total;
            sub.Processed_Companies__c = completed;
            sub.Failed_Companies__c    = failed;
            sub.Completed_At__c        = DateTime.now();

            // Determine final status
            if (failed == 0) {
                sub.Status__c = 'completed';
            } else if (completed > 0) {
                sub.Status__c = 'partial';
            } else {
                sub.Status__c = 'failed';
            }

            toUpdate.add(sub);
        }

        updateRecords(toUpdate, 'SubmissionProgressUpdater.processUploadComplete');
    }

    // ── BATCH_PROGRESS ───────────────────────────────────────────────────────
    public static void processBatchProgress(List<AI_Navigator_Inbound__e> events) {
        Map<String, AI_Navigator_Inbound__e> subIdToEvent = mapBySubmissionId(events);
        if (subIdToEvent.isEmpty()) return;

        List<Submission__c> toUpdate = new List<Submission__c>();

        for (Submission__c sub : [
            SELECT Id, Submission_Id__c, Status__c,
                   Total_Companies__c, Processed_Companies__c,
                   Failed_Companies__c
            FROM Submission__c
            WHERE Submission_Id__c IN :subIdToEvent.keySet()
        ]) {
            AI_Navigator_Inbound__e evt = subIdToEvent.get(sub.Submission_Id__c);
            Map<String, Object> payload = parsePayload(evt.Payload__c);

            sub.Total_Companies__c     = intVal(payload.get('total'));
            sub.Processed_Companies__c = intVal(payload.get('processed'));
            sub.Failed_Companies__c    = intVal(payload.get('failed'));
            sub.Status__c              = 'processing';
            toUpdate.add(sub);
        }

        updateRecords(toUpdate, 'SubmissionProgressUpdater.processBatchProgress');
    }

    // ── Private Helpers ──────────────────────────────────────────────────────
    private static Map<String, AI_Navigator_Inbound__e> mapBySubmissionId(
        List<AI_Navigator_Inbound__e> events
    ) {
        Map<String, AI_Navigator_Inbound__e> result = new Map<String, AI_Navigator_Inbound__e>();
        for (AI_Navigator_Inbound__e evt : events) {
            Map<String, Object> payload = parsePayload(evt.Payload__c);
            String submissionId = (String) payload.get('submissionId');
            if (String.isNotBlank(submissionId)) {
                result.put(submissionId, evt);
            }
        }
        return result;
    }

    private static Map<String, Object> parsePayload(String payloadJson) {
        if (String.isBlank(payloadJson)) return new Map<String, Object>();
        try {
            return (Map<String, Object>) JSON.deserializeUntyped(payloadJson);
        } catch (Exception e) {
            System.debug(LoggingLevel.ERROR,
                'SubmissionProgressUpdater: Failed to parse payload. ' + e.getMessage());
            return new Map<String, Object>();
        }
    }

    private static Integer intVal(Object val) {
        if (val == null) return 0;
        return Integer.valueOf(String.valueOf(val));
    }

    private static void updateRecords(List<Submission__c> records, String context) {

        } catch (DmlException de) {

        }
    }

 *                ordered newest-first, limit 200.
 * @author        AI Navigator Team
 * @last modified : 2026-04-22

public with sharing class SubmissionsController {

    /**
     * Returns serialised list of Submission__c records for the current user.
     * Shape mirrors the original mock so submissions.js needs zero changes.
     *
     * Fields mapped:
     *   submissionId  → Submission_Id__c
     *   fileName      → File_Name__c
     *   status        → Status__c          (pending|processing|completed|failed|partial)
     *   totalCompanies→ Total_Companies__c
     *   processed     → Processed_Companies__c
     *   failed        → Failed_Companies__c
     *   submittedAt   → Submitted_At__c
     *   source        → derived: 'Upload' when File_Name__c is present, else 'Input'
     */
    @AuraEnabled(cacheable=true)
    public static String getSubmissions() {
        try {
            List<Submission__c> records = [
                SELECT  Id,
                        Submission_Id__c,
                        File_Name__c,
                        Status__c,
                        Total_Companies__c,
                        Processed_Companies__c,
                        Failed_Companies__c,
                        Submitted_At__c,
                        Completed_At__c,
                        Submitted_By__c
                FROM    Submission__c
                WHERE   Submitted_By__c = :UserInfo.getUserId()
                ORDER BY Submitted_At__c DESC
                LIMIT   200
            ];

            List<Map<String, Object>> result = new List<Map<String, Object>>();

            for (Submission__c s : records) {
                // Derive source: a file name means it came from the Upload LWC.
                String source = String.isNotBlank(s.File_Name__c) ? 'Upload' : 'Input';

                // Progress percentage — safe divide
                Integer total     = s.Total_Companies__c     != null ? s.Total_Companies__c.intValue()     : 0;
                Integer processed = s.Processed_Companies__c != null ? s.Processed_Companies__c.intValue() : 0;
                Integer failed    = s.Failed_Companies__c    != null ? s.Failed_Companies__c.intValue()    : 0;

                Integer progressPct = (total > 0)
                    ? Math.min(100, Math.roundToLong(((Decimal) processed / total) * 100).intValue())
                    : 0;

                // Completion detection for BATCHPROGRESS-only flows:
                // when Azure has not yet fired UPLOADPROCESSED we can derive
                // isComplete from the counts.
                Boolean isComplete = (total > 0) && ((processed + failed) >= total);

                result.add(new Map<String, Object>{
                    'submissionId'      => s.Submission_Id__c,
                    'salesforceId'      => s.Id,
                    'fileName'          => s.File_Name__c,
                    'source'            => source,
                    'status'            => s.Status__c,
                    'totalCompanies'    => total,
                    'processed'         => processed,
                    'failed'            => failed,
                    'progressPct'       => progressPct,
                    'isComplete'        => isComplete,
                    'submittedAt'       => s.Submitted_At__c != null
                                            ? s.Submitted_At__c.formatGmt('yyyy-MM-dd\'T\'HH:mm:ss\'Z\'')
                                            : null,
                    'completedAt'       => s.Completed_At__c != null
                                            ? s.Completed_At__c.formatGmt('yyyy-MM-dd\'T\'HH:mm:ss\'Z\'')
                                            : null
                });
            }

            return JSON.serialize(result);

        } catch (Exception e) {
            throw new AuraHandledException('Error fetching submissions: ' + e.getMessage());
        }
    }
}

        try {
            List<Submission__c> subs = [
                SELECT Id, Submission_Id__c, File_Name__c, Status__c,
                       Submitted_At__c, Submitted_By__c, Total_Companies__c,
                       Processed_Companies__c, Failed_Companies__c
                FROM Submission__c
                WHERE Submitted_By__c = :UserInfo.getUserId()
                ORDER BY Submitted_At__c DESC
                LIMIT 50
            ];

            List<Map<String, Object>> result = new List<Map<String, Object>>();
            for (Submission__c s : subs) {
                result.add(new Map<String, Object>{
                    'submissionId' => s.Submission_Id__c,
                    'recordId'     => s.Id,
                    'fileName'     => s.File_Name__c,
                    'status'       => s.Status__c,
                    'submittedAt'  => String.valueOf(s.Submitted_At__c),
                    'total'        => s.Total_Companies__c,
                    'processed'    => s.Processed_Companies__c,
                    'failed'       => s.Failed_Companies__c
                });
            }
            return JSON.serialize(result);
        } catch (Exception e) {
            throw new AuraHandledException('Error loading submissions: ' + e.getMessage());
        }
    }

    @AuraEnabled
    public static String uploadFile(
        String fileName,
        String base64Data,
        String defaultStage,
        String defaultClientType
    ) {
        String correlationId  = null;
        String submissionId   = null;
        Map<String, Object> payload = null;

        try {
            // ── Guard: daily limit check ────────────────────────────────────
            Integer usedToday = getTodayUploadCount();
            if (usedToday >= DAILY_LIMIT) {
                throw new AuraHandledException(
                    'Daily upload limit of ' + DAILY_LIMIT + ' reached. Try again tomorrow.'
                );
            }

            // ── 1. Generate IDs ─────────────────────────────────────────────
            correlationId = AiNavigatorEventService.generateCorrelationId();
            submissionId  = correlationId;

            // ── 2. Calculate file size from base64 ──────────────────────────
            Integer fileSize = (Integer)(base64Data.length() * 0.75);

            // ── 3. Create Submission__c record ──────────────────────────────
            Submission__c sub = new Submission__c(
                Submission_Id__c = submissionId,
                File_Name__c     = fileName,
                Status__c        = 'processing',
                Submitted_By__c  = UserInfo.getUserId(),
                Submitted_At__c  = DateTime.now()
            );
            insert sub;
        // ── 3b. Parse CSV and pre-create Company__c records ─────────────────
        List<Company__c> companiesToInsert = new List<Company__c>();
        List<Map<String, String>> parsedRows = parseCsv(base64Data);

        for (Map<String, String> row : parsedRows) {
            companiesToInsert.add(new Company__c(
                Name             = row.get('Name'),
                Website__c       = row.get('Website'),
                City__c          = row.get('City'),
                State__c         = row.get('State'),
                Stage__c         = String.isNotBlank(row.get('Stage'))      ? row.get('Stage')      : defaultStage,
                Client_Type__c   = String.isNotBlank(row.get('ClientType')) ? row.get('ClientType') : defaultClientType,
                Status__c        = 'pending',
                Correlation_Id__c = correlationId
            ));
        }

        if (!companiesToInsert.isEmpty()) {
            insert companiesToInsert;
        }

        // ── 3c. Build company list WITH real SF IDs for the payload ──────────
        List<Map<String, Object>> companyList = new List<Map<String, Object>>();
        for (Company__c c : companiesToInsert) {
            companyList.add(new Map<String, Object>{
                'salesforceId' => c.Id,          // ← Azure echoes this back in callbacks
                'name'         => c.Name,
                'website'      => c.Website__c,
                'city'         => c.City__c,
                'state'        => c.State__c,
                'stage'        => c.Stage__c,
                'clientType'   => c.Client_Type__c
            });
        }         

            // ── 4. Build payload ─────────────────────────────────────────────
            payload = new Map<String, Object>{
                'fileName'     => fileName,
                'fileContent'  => base64Data,
                'submissionId' => submissionId,
                'companies'     => companyList,      // ← ADD THIS — real SF IDs included
                'stage'        => defaultStage,
                'clientType'   => defaultClientType,
                'fileSize'     => fileSize,
                'submittedBy'  => UserInfo.getUserId(),
                'submittedAt'  => String.valueOf(DateTime.now())
            };

            // ── 5. Enqueue async HTTP callout to Azure ───────────────────────────
                System.enqueueJob(new AzureUploadCompanyCallout(
                    sub.Id,
                    JSON.serialize(payload),
                    correlationId
                ));

            // ── 6. Log the outbound event (SUCCESS) ──────────────────────────
            writeLog('PLATFORM_EVENT', 'UploadController.uploadFile',
                    JSON.serialize(payload), null, 'Success', 202,       // ← Integer 202
                    correlationId, 'submissionId=' + submissionId + ' file=' + fileName);

            // ── 7. Return to LWC ─────────────────────────────────────────────
            Integer newUsedToday = usedToday + 1;
            return JSON.serialize(new Map<String, Object>{
                'success'       => true,
                'message'       => 'File uploaded successfully. Processing will start shortly.',
                'submissionId'  => submissionId,
                'recordId'      => sub.Id,
                'fileName'      => fileName,
                'fileSize'      => fileSize,
                'correlationId' => correlationId,
                'dailyLimit'    => DAILY_LIMIT,
                'usedToday'     => newUsedToday,
                'remaining'     => Math.max(0, DAILY_LIMIT - newUsedToday)
            });

        } catch (DmlException dmlEx) {
            System.debug(LoggingLevel.ERROR, 'UploadController DML error: ' + dmlEx.getMessage());
            writeLog('PLATFORM_EVENT', 'UploadController.uploadFile',
                payload != null ? JSON.serialize(payload) : null,
                null, 'Failed', 500,                                 // ← Integer 500
                correlationId, 'DML ERROR: ' + dmlEx.getDmlMessage(0));
            throw new AuraHandledException(
                'Could not save the submission record. ' + dmlEx.getDmlMessage(0)
            );

        } catch (AuraHandledException ahe) {
            // Daily limit or validation — log it then re-throw
           writeLog('PLATFORM_EVENT', 'UploadController.uploadFile',
                null, null, 'Failed', 400,                           // ← Integer 400
                correlationId, 'VALIDATION: ' + ahe.getMessage());
            throw ahe;

        } catch (Exception e) {
            System.debug(LoggingLevel.ERROR, 'UploadController error: ' + e.getMessage());
            writeLog('PLATFORM_EVENT', 'UploadController.uploadFile',
                payload != null ? JSON.serialize(payload) : null,
                null, 'Failed', 500,                                 // ← Integer 500
                correlationId, 'EXCEPTION: ' + e.getMessage());
            throw new AuraHandledException('An unexpected error occurred. Please try again.');
        }
    }

    // ── Private log helper ───────────────────────────────────────────────────────
   private static void writeLog(
    String httpMethod,
    String endpointName,
    String requestBody,
    String responseBody,
    String status,
    Integer httpStatusCode,             // ← was String, now Integer
    String correlationId,
    String errorMessage
    ) {
        try {
            if (requestBody != null) {
                Map<String, Object> bodyMap = (Map<String, Object>) JSON.deserializeUntyped(requestBody);
                bodyMap.remove('fileContent');
                requestBody = JSON.serialize(bodyMap);
            }

            if (requestBody != null && requestBody.length() > 32000) {
                requestBody = requestBody.substring(0, 32000) + '...[TRUNCATED]';
            }

            insert new Azure_Request_Log__c(
                HTTP_Method__c      = httpMethod,
                Endpoint_Name__c    = endpointName,
                Request_Body__c     = requestBody,
                Response_Body__c    = responseBody,
                Status__c           = 'Success',
                HTTP_Status_Code__c = httpStatusCode,   // ← Integer, matches Number field
                Correlation_ID__c   = correlationId,
                Error_Message__c    = errorMessage
            );

        } catch (Exception logEx) {
            try {
                insert new Azure_Request_Log__c(
                    HTTP_Method__c      = httpMethod,
                    Endpoint_Name__c    = endpointName,
                    Status__c           = 'Failed',
                    HTTP_Status_Code__c = httpStatusCode,   // ← Integer here too
                    Correlation_ID__c   = correlationId,
                    Error_Message__c    = 'writeLog INSERT FAILED: ' + logEx.getMessage()
                                        + ' | Original error: ' + errorMessage
                );
            } catch (Exception fallbackEx) {
                // Last resort — only two fields, nothing that can fail
    insert new Azure_Request_Log__c(
        Status__c        = 'Failed',
        Error_Message__c = 'writeLog ALL inserts failed.'
                            + ' Final: '    + fallbackEx.getMessage()
                            + ' Fallback: ' + fallbackEx.getMessage()
                            + ' Primary: '  + fallbackEx.getMessage()
                            + ' Original: ' + errorMessage
        );
            }
        }
    }

        /**
         * Counts how many Submission__c records the current user has created today.
         * Used for daily limit enforcement.
         */
        private static Integer getTodayUploadCount() {
            DateTime startOfDay = DateTime.newInstance(Date.today(), Time.newInstance(0, 0, 0, 0));
            return [
                SELECT COUNT()
                FROM Submission__c
                WHERE Submitted_By__c = :UserInfo.getUserId()
                AND   Submitted_At__c >= :startOfDay
            ];
        }
    // Normalize header names to match row.get() keys
    private static final Map<String, String> HEADER_MAP = new Map<String, String>{
        'company name' => 'Name',
        'name'         => 'Name',
        'company url'  => 'Website',
        'website'      => 'Website',
        'url'          => 'Website',
        'city'         => 'City',
        'state'        => 'State',
        'stage'        => 'Stage',
        'client type'  => 'ClientType',
        'clienttype'   => 'ClientType'
    };

    private static List<Map<String, String>> parseCsv(String base64Data) {
        if (base64Data.contains(',')) {
            base64Data = base64Data.substringAfter(',');
        }

        String csvContent = EncodingUtil.base64Decode(base64Data).toString();
        List<Map<String, String>> rows = new List<Map<String, String>>();
        List<String> lines = csvContent.split('\n');

        if (lines.isEmpty()) return rows;

        // Parse + normalize header row
        List<String> headers = new List<String>();
        for (String h : lines[0].replace('\r', '').split(',')) {
            String normalized = HEADER_MAP.get(h.trim().toLowerCase());
            headers.add(normalized != null ? normalized : h.trim());
        }

        // Parse data rows
        for (Integer i = 1; i < lines.size(); i++) {
            String line = lines[i].replace('\r', '').trim();
            if (String.isBlank(line)) continue;

            List<String> values = line.split(',');
            Map<String, String> row = new Map<String, String>();
            for (Integer j = 0; j < headers.size(); j++) {
                row.put(headers[j], j < values.size() ? values[j].trim() : '');
            }
            rows.add(row);
        }
        return rows;
    }
