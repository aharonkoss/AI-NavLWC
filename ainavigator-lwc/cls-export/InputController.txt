/**
 * @description Controller for the inputForm LWC.
 *              Handles single-company manual submission:
 *              1. Creates a Company__c record (Status = processing)
 *              2. Publishes AINavigatorOutbound__e (RESEARCHREQUEST)
 *              3. Returns the real Salesforce record ID to the LWC
 *
 * @group       AI Navigator - LWC Controllers
 * @last-modified 2026-04-21
 */
public with sharing class InputController {

    @AuraEnabled
    public static String submitCompany(
        String companyName,
        String website,
        String city,
        String state,
        String stage,
        String clientType
    ) {
        try {
            // ── 1. Generate correlation ID ──────────────────────────────────
            String correlationId = AiNavigatorEventService.generateCorrelationId();

            // ── 2. Create Company__c record ─────────────────────────────────
            Company__c company = new Company__c(
                Name             = companyName,
                Website__c       = website,
                City__c          = city,
                State__c         = state,
                Stage__c         = stage,
                Client_Type__c   = clientType,
                Status__c        = 'processing',
                Correlation_Id__c = correlationId,
                Assigned_User__c = UserInfo.getUserId()
            );
            insert company;

            // ── 3. Build versioned payload ──────────────────────────────────
            // AZURE-Q5: Confirm SF owns correlationId generation.
            // AZURE-Q4: Confirm channel = /event/AINavigatorOutbound__e
            Map<String, Object> payload = new Map<String, Object>{
                'companyId'   => company.Id,
                'name'        => companyName,
                'website'     => website,
                'city'        => city,
                'state'       => state,
                'stage'       => stage,
                'clientType'  => clientType,
                'submittedBy' => UserInfo.getUserId(),
                'submittedAt' => String.valueOf(DateTime.now())
                // AZURE-Q2: Add any additional fields Azure needs for RESEARCHREQUEST here
                //           without changing schema — just add keys to this map.
            };

            // ── 4. Publish outbound Platform Event ──────────────────────────
            AiNavigatorEventService.publish(
                'RESEARCHREQUEST',
                'ResearchRequest.v1',
                payload,
                correlationId
            );

            // ── 5. Return real record data to LWC ───────────────────────────
            return JSON.serialize(new Map<String, Object>{
                'success'       => true,
                'companyId'     => company.Id,
                'correlationId' => correlationId,
                'name'          => companyName,
                'website'       => website,
                'city'          => city,
                'state'         => state,
                'stage'         => stage,
                'clientType'    => clientType,
                'status'        => 'processing',
                'createdAt'     => String.valueOf(DateTime.now())
            });

        } catch (DmlException dmlEx) {
            System.debug(LoggingLevel.ERROR,
                'InputController.submitCompany DML error: ' + dmlEx.getMessage());
            throw new AuraHandledException(
                'Could not save the company record. ' + dmlEx.getDmlMessage(0)
            );
        } catch (AuraHandledException ahe) {
            throw ahe; // re-throw event publish failures as-is
        } catch (Exception e) {
            System.debug(LoggingLevel.ERROR,
                'InputController.submitCompany unexpected error: ' + e.getMessage());
            throw new AuraHandledException(
                'An unexpected error occurred. Please try again.'
            );
        }
    }
}
