/**
 * @description Shared HTTP utility for all outbound Azure APIM callouts.
 *              Builds pre-configured HttpRequest objects using headers pulled
 *              from AI_Navigator_Config__mdt (DeveloperName = 'Dev').
 *
 *              Required Azure headers:
 *                - Ocp-Apim-Subscription-Key  ← new ApimSubscriptionKey__c field
 *                - X-Client-Id                ← Client_Id__c from CMT
 *                - X-User-Id                  ← SF User Id of submitting user
 *                - X-Correlation-Id           ← UUID generated per request
 *                - Content-Type application/json
 *                - Authorization Bearer       ← injected by Named Credential
 */
public without sharing class AzureHttpService {

    private static final Integer DEFAULT_TIMEOUT_MS = 60000;

    /**
     * Build a ready-to-send HttpRequest with all required Azure headers.
     *
     * @param method           HTTP verb: POST, GET, PUT
     * @param path             Path appended to Named Credential base URL
     *                         e.g. /api/v1/user/companies/manual
     * @param correlationId    UUID generated by the caller
     * @param submittingUserId Salesforce User Id of the user who triggered the action
     * @return Fully configured HttpRequest — caller sets body if needed, then sends
     */
    public static HttpRequest buildRequest(
        String method,
        String path,
        String correlationId,
        String submittingUserId
    ) {
        AI_Navigator_Config__mdt config = loadConfig();

        HttpRequest req = new HttpRequest();

        // Named Credential handles Authorization: Bearer <JWT> automatically
        req.setEndpoint('callout:' + config.Named_Credential_Name__c + path);
        req.setMethod(method);
        req.setTimeout(
            config.Callout_Timeout_Ms__c != null
                ? Integer.valueOf(config.Callout_Timeout_Ms__c)
                : DEFAULT_TIMEOUT_MS
        );

        // Required headers — matched to Azure APIM contract
        req.setHeader('Content-Type',               'application/json');
        req.setHeader('Accept',                     'application/json');
        req.setHeader('Ocp-Apim-Subscription-Key',  config.ApimSubscriptionKey__c);
        req.setHeader('X-Client-Id',                config.Client_Id__c);
        req.setHeader('X-User-Id',
            String.isNotBlank(submittingUserId) ? submittingUserId : UserInfo.getUserId()
        );
        req.setHeader('X-Correlation-Id',           correlationId);

        return req;
    }

    /**
     * Parse a standard Azure error response envelope and return a readable message.
     */
    public static String parseErrorResponse(HttpResponse res) {
        try {
            Map<String, Object> body =
                (Map<String, Object>) JSON.deserializeUntyped(res.getBody());
            String msg    = (String) body.get('message');
            String code   = (String) body.get('code');
            String corrId = (String) body.get('correlationId');
            return (code   != null ? '[' + code   + '] ' : '')
                 + (msg    != null ? msg            : res.getBody())
                 + (corrId != null ? ' (correlationId: ' + corrId + ')' : '');
        } catch (Exception e) {
            return 'HTTP ' + res.getStatusCode() + ' — ' + res.getBody();
        }
    }

    // ── Private ──────────────────────────────────────────────────────────────

    private static AI_Navigator_Config__mdt loadConfig() {
        List<AI_Navigator_Config__mdt> configs = [
            SELECT Named_Credential_Name__c,
                   ApimSubscriptionKey__c,
                   Client_Id__c,
                   Callout_Timeout_Ms__c,
                   Is_Active__c
            FROM   AI_Navigator_Config__mdt
            WHERE  DeveloperName = 'Dev'
            LIMIT  1
        ];

        if (configs.isEmpty() || configs[0].Named_Credential_Name__c == null) {
            throw new AzureHttpServiceException(
                'AI_Navigator_Config__mdt "Dev" record is missing or incomplete.'
            );
        }
        if (!configs[0].Is_Active__c) {
            throw new AzureHttpServiceException(
                'Azure callouts are disabled via kill switch (Is_Active__c = false).'
            );
        }
        return configs[0];
    }

    public class AzureHttpServiceException extends Exception {}
}
